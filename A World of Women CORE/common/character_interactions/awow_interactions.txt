awow_flag_exclude_interaction = {
    category = interaction_category_diplomacy
    icon = vassal_claim_liege_title_interaction
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes
    common_interaction = yes
    auto_accept = yes


    is_shown = {
        scope:recipient = {
            is_alive = yes
            is_female = no
            NOT = { has_character_flag = awow_excluded_from_culling }
        }
    }

    is_valid = {
        scope:recipient = {
            is_alive = yes
            is_female = no
            NOT = { has_character_flag = awow_excluded_from_culling }
        }
    }

    on_accept = {
        scope:recipient = {
            add_character_flag = awow_excluded_from_culling
            set_variable = {
                name = awow_culling_protection_owner
                value = scope:actor
            }
        }
    }
}

awow_flag_include_interaction = {
    category = interaction_category_diplomacy
    icon = debug_chars
    common_interaction = yes
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes
    common_interaction = yes
    auto_accept = yes

    is_shown = {
        scope:recipient = {
            is_alive = yes
            is_female = no
            has_character_flag = awow_excluded_from_culling
            awow_can_revoke_culling_protection_trigger = yes
        }
    }

    is_valid = {
        scope:recipient = {
            is_alive = yes
            is_female = no
            has_character_flag = awow_excluded_from_culling
            awow_can_revoke_culling_protection_trigger = yes
        }
    }

    on_accept = {
        scope:recipient = {
            remove_character_flag = awow_excluded_from_culling
            remove_variable = awow_culling_protection_owner
        }
    }
}
