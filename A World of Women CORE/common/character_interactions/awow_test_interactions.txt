# TEST interaction: queue a same-sex conception attempt with the chosen partner.
# It now queues the production carrier-selection and delayed-conception path,
# rather than directly starting a pregnancy or bypassing fertility.
awow_test_same_sex_pregnancy_interaction = {
    category = interaction_category_uncategorized
    icon = debug_chars
    common_interaction = yes
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes
    auto_accept = yes

    is_shown = {
        debug_only = yes
        scope:actor = {
            is_female = yes
        }
        scope:recipient = {
            is_alive = yes
            is_female = yes
            OR = {
                is_spouse_of = scope:actor
                is_concubine_of = scope:actor
            }
        }
    }

    is_valid = {
        scope:actor = {
            awow_same_sex_can_conceive_with_trigger = {
                PARTNER = scope:recipient
            }
        }
    }

    on_accept = {
        debug_log = "AWOW SS-PREG TEST: queued the production conception path"
        scope:actor = {
            save_scope_as = awow_root
        }
        scope:recipient = {
            save_scope_as = awow_partner
            awow_decide_same_sex_mother_effect = yes
        }
        clear_saved_scope = awow_partner
        clear_saved_scope = awow_root
    }
}

awow_test_has_relation = {
    category = interaction_category_uncategorized
    icon = debug_chars
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes
    common_interaction = yes
    auto_accept = yes

    is_shown = {
        debug_only = yes
        scope:recipient = {
            is_alive = yes
            OR = {
                has_relation_awow_parent = scope:actor
                has_relation_awow_child = scope:actor
                has_relation_awow_sibling = scope:actor
                has_relation_awow_grandchild = scope:actor
                has_relation_awow_grandparent = scope:actor
            }
        }
    }

    is_valid = {
        scope:recipient = {
            is_alive = yes
            OR = {
                has_relation_awow_parent = scope:actor
                has_relation_awow_child = scope:actor
                has_relation_awow_sibling = scope:actor
                has_relation_awow_grandchild = scope:actor
                has_relation_awow_grandparent = scope:actor
            }
        }
    }

    on_accept = {
        scope:actor = {
            if = {
                limit = {
                    has_relation_awow_child = scope:recipient
                }
                add_gold = 10
            }
            if = {
                limit = {
                    has_relation_awow_parent = scope:recipient
                }
                add_gold = 100
            }
            if = {
                limit = {
                    has_relation_awow_sibling = scope:recipient
                }
                add_gold = 50
            }
            if = {
                limit = {
                    has_relation_awow_grandchild = scope:recipient
                }
                add_gold = 5
            }
            if = {
                limit = {
                    has_relation_awow_grandparent = scope:recipient
                }
                add_gold = 1000
            }
        }
    }
}
