# Debug test for AWOW's production carrier selection and fertility roll.
awow_test_native_pregnancy_production_interaction = {
    category = interaction_category_uncategorized
    icon = debug_chars
    common_interaction = yes
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes
    auto_accept = yes

    is_shown = {
        debug_only = yes
        scope:actor = { is_female = yes }
        scope:recipient = {
            is_alive = yes
            is_female = yes
            OR = {
                is_spouse_of = scope:actor
                is_concubine_of = scope:actor
                scope:actor = { is_concubine_of = prev }
                has_relation_lover = scope:actor
                has_relation_soulmate = scope:actor
            }
        }
    }

    is_valid = {
        scope:actor = {
            awow_same_sex_can_conceive_with_trigger = {
                PARTNER = scope:recipient
            }
        }
    }

    on_accept = {
        scope:actor = { save_scope_as = awow_root }
        scope:recipient = {
            save_scope_as = awow_partner
            awow_decide_same_sex_mother_effect = yes
        }
        clear_saved_scope = awow_partner
        clear_saved_scope = awow_root
    }
}
