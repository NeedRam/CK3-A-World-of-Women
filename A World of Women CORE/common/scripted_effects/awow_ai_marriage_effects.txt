# AWOW AI ruler marriage pulse effect.
# Scoped to an unmarried adult AI ruler.
# Attempts to find a spouse through multiple diplomatic channels:

#########################################################
########### ----------- MARRIAGES ----------- ###########
#########################################################

#######################################
# --------- Ruler Marriage --------- #
#######################################

awow_ai_ruler_find_spouse_effect = {
    save_scope_as = awow_ruler
    save_scope_as = awow_single_gal

    awow_find_spouse_effect = yes

    clear_saved_scope = awow_single_gal
    clear_saved_scope = awow_ruler
}

################################################################
# --------- Ruler Diplomatic Adult Children Marriage --------- #
################################################################

awow_ai_ruler_find_spouse_for_child_effect = {
    save_scope_as = awow_ruler
    random_child = {
        limit = {
            is_adult = yes
            awow_can_marry_trigger = yes
        }
        save_scope_as = awow_single_gal
        awow_find_spouse_effect = yes
        clear_saved_scope = awow_single_gal

    }
    clear_saved_scope = awow_ruler
}

###############################
# --------- Helpers --------- #
###############################

# --------- Spouse Finder --------- #
awow_find_spouse_effect = {

    # Try Ruler > Child > Family
    # Prefer Realm / Ally
    every_living_character = {
        limit = {
            this != root
            is_adult = yes
            in_diplomatic_range = scope:awow_ruler
            awow_can_marry_trigger = yes
            is_playable_character = yes
        }
        save_scope_as = awow_other_ruler
        awow_find_spouse_walk_effect = yes
        clear_saved_scope = awow_other_ruler

    }

    if = {
        limit = { is_married = no }

        # Try a Courtier
        random_courtier = {
            limit = {
                is_adult = yes
                awow_can_marry_trigger = yes
            }
            save_scope_as = awow_potential_spouse
            random = {
                chance = 5
                marry_same_sex = scope:awow_potential_spouse
            }
            clear_saved_scope = awow_potential_spouse
        }
    }
}

# --------- Walk through Ruler > Children > Family --------- #
awow_find_spouse_walk_effect = {
    awow_find_spouse_try_ruler_effect = yes
    awow_find_spouse_try_child_of_ruler_effect = yes
    awow_find_spouse_try_family_of_ruler_effect = yes
}

# Ruler
awow_find_spouse_try_ruler_effect = {
    save_scope_as = awow_potential_spouse
    random = {
        chance = {
            value = 1
            if = {
                limit = {
                    OR = { # in realm/ally
                        top_liege = scope:awow_ruler.top_liege
                        is_allied_to = scope:awow_ruler
                    }
                }
                add = 0.5
            }
        }
        awow_try_marriage_interaction_effect = yes
    }
    clear_saved_scope = awow_potential_spouse
}

# Children
awow_find_spouse_try_child_of_ruler_effect = {

    # Try their child
    if = {
        limit = {
            scope:awow_ruler = { is_married = no }
            any_child = {
                is_adult = yes
                awow_can_marry_trigger = yes
            }
        }
        random_child = {
            limit = {
                is_adult = yes
                awow_can_marry_trigger = yes
            }
            save_scope_as = awow_potential_spouse
            random = {
                chance = {
                    value = 1
                    if = {
                        limit = {
                            OR = { # in realm/ally
                                top_liege = scope:awow_ruler.top_liege
                                is_allied_to = scope:awow_ruler
                            }
                        }
                        add = 1
                    }
                }
                awow_try_marriage_interaction_effect = yes
            }
            clear_saved_scope = awow_potential_spouse
        }
    }
}

# Family
awow_find_spouse_try_family_of_ruler_effect = {

    # Try their family
    if = {
        limit = {
            scope:awow_ruler = { is_married = no }
            any_close_family_member = {
                is_adult = yes
                awow_can_marry_trigger = yes
            }
        }
        random_close_family_member = {
            limit = {
                is_adult = yes
                awow_can_marry_trigger = yes
            }
            save_scope_as = awow_potential_spouse
            random = {
                chance = {
                    value = 0.25
                    if = {
                        limit = {
                            OR = { # in realm/ally
                                top_liege = scope:awow_ruler.top_liege
                                is_allied_to = scope:awow_ruler
                            }
                        }
                        add = 0.75
                    }
                }
                awow_try_marriage_interaction_effect = yes
            }
            clear_saved_scope = awow_potential_spouse
        }
    }
}

# --------- Marriage Interaction --------- #
awow_try_marriage_interaction_effect = {
    run_interaction = {
        interaction = arrange_marriage_interaction
        actor = scope:awow_ruler
        recipient = scope:awow_other_ruler
        secondary_actor = scope:awow_single_gal
        secondary_recipient = scope:awow_potential_spouse
        execute_threshold = accept
    }
}

awow_betrothe_child_effect = {
    save_scope_as = awow_ruler
    random_child = {
        limit = {
            is_adult = no
            awow_can_marry_trigger = yes
        }
        save_scope_as = alliance_fodder
        awow_find_betrothal_effect = yes
        clear_saved_scope = alliance_fodder
    }
    clear_saved_scope = awow_ruler
}

awow_find_betrothal_effect = {

    # Try Ruler > Child > Family
    every_living_character = {
        limit = {
            this != root
            in_diplomatic_range = scope:awow_ruler
            awow_can_marry_trigger = yes
            is_playable_character = yes
            NOT = { is_allied_to = scope:awow_ruler }
        }
        save_scope_as = awow_other_ruler
        awow_find_betrothal_walk_effect = yes
        clear_saved_scope = awow_other_ruler
    }
}

# --------- Walk through Ruler > Children > Family --------- #
awow_find_betrothal_walk_effect = {
    awow_find_betrothal_try_ruler_effect = yes
    awow_find_betrothal_try_child_of_ruler_effect = yes
    awow_find_betrothal_try_family_of_ruler_effect = yes
}

# Ruler
awow_find_betrothal_try_ruler_effect = {
    save_scope_as = awow_potential_betrothal
    random = {
        chance = 1
        awow_same_sex_betrothal_effect = yes
    }
    clear_saved_scope = awow_potential_betrothal
}

# Children
awow_find_spouse_try_child_of_ruler_effect = {

    # Try their child
    if = {
        limit = {
            scope:alliance_fodder = { NOT = { has_character_flag = awow_same_sex_betrothal } }
            any_child = { awow_can_marry_trigger = yes }
        }
        random_child = {
            limit = { awow_can_marry_trigger = yes }
            save_scope_as = awow_potential_betrothal
            random = {
                chance = 0.4
                awow_same_sex_betrothal_effect = yes
            }
            clear_saved_scope = awow_potential_betrothal
        }
    }
}

# Family
awow_find_spouse_try_family_of_ruler_effect = {

    # Try their family
    if = {
        limit = {
            scope:alliance_fodder = { NOT = { has_character_flag = awow_same_sex_betrothal } }
            any_close_family_member = {
                awow_can_marry_trigger = yes
            }
        }
        random_close_family_member = {
            limit = { awow_can_marry_trigger = yes }
            save_scope_as = awow_potential_spouse
            random = {
                chance = 0.1
                awow_same_sex_betrothal_effect = yes
            }
            clear_saved_scope = awow_potential_spouse
        }
    }
}

# --------- Betrothal Effect --------- #
awow_same_sex_betrothal_effect = {
    awow_arrange_betrothal_effect = yes
    awow_create_betrothal_alliance_effect = yes
}

awow_arrange_betrothal_effect = {
    scope:alliance_fodder = {
        set_variable = {
            name = awow_betrothed_to
            value = scope:awow_potential_betrothal
        }
        add_character_flag = awow_same_sex_betrothal
        add_opinion = {
            target = scope:awow_potential_betrothal
            modifier = awow_betrothed_opinion
        }
    }
    scope:awow_potential_betrothal = {
        set_variable = {
            name = awow_betrothed_to
            value = scope:alliance_fodder
        }
        add_character_flag = awow_same_sex_betrothal
        add_opinion = {
            target = scope:alliance_fodder
            modifier = awow_betrothed_opinion
        }
    }
}

# Create alliance between the two rulers who arranged the betrothal
awow_create_betrothal_alliance_effect = {
    scope:awow_ruler = {
        if = {
            limit = {
                scope:awow_other_ruler = { is_alive = yes }
                NOT = { is_allied_to = scope:awow_other_ruler }
            }
            create_alliance = {
                target = scope:awow_other_ruler
                allied_through_owner = scope:alliance_fodder
                allied_through_target = scope:awow_potential_betrothal
            }
        }
    }

}
