# AWOW AI marriage effects for unmarried adult rulers and their children.
#########################################################
########### ----------- MARRIAGES ----------- ###########
#########################################################

awow_ai_matchmaking_effect = {
    if = {
        limit = {
            is_ai = yes
            is_adult = yes
            is_playable_character = yes
            OR = {
                is_landless_adventurer = yes
                AND = {
                    is_landed = yes
                    highest_held_title_tier >= tier_county
                }
            }
        }
        trigger_event = awow_ai_marriage.0001
        trigger_event = awow_ai_marriage.0002
    }
}

awow_ai_matchmaking_burst_effect = {
    every_living_character = {
        awow_ai_matchmaking_effect = yes
    }
}

#######################################
# --------- Ruler Marriage --------- #
#######################################

awow_ai_ruler_find_spouse_effect = {
    save_scope_as = awow_ruler
    save_scope_as = awow_single_gal

    awow_find_spouse_effect = yes

    clear_saved_scope = awow_single_gal
    clear_saved_scope = awow_ruler
}

################################################################
# --------- Ruler Diplomatic Adult Children Marriage --------- #
################################################################

awow_ai_ruler_find_spouse_for_child_effect = {
    save_scope_as = awow_ruler
    random_child = {
        limit = {
            is_ai = yes
            is_adult = yes
            awow_can_marry_trigger = yes
        }
        save_scope_as = awow_single_gal
        awow_find_spouse_effect = yes
        clear_saved_scope = awow_single_gal

    }
    clear_saved_scope = awow_ruler
}

# --------- Spouse Finder --------- #
awow_find_spouse_effect = {
    awow_collect_ruler_courtier_potential_spouses_effect = yes
    awow_build_weighted_potential_spouse_pool_effect = yes
    awow_select_weighted_potential_spouse_effect = yes
}

# --------- Weight Builder --------- #
awow_collect_ruler_courtier_potential_spouses_effect = {
    scope:awow_ruler = {
        save_scope_as = awow_other_ruler
        every_courtier = {
            limit = { awow_weighted_potential_spouse_trigger = yes }
            awow_store_tertiary_potential_spouse_effect = yes
        }
        clear_saved_scope = awow_other_ruler
    }
}

awow_build_weighted_potential_spouse_pool_effect = {
    every_ruler = {
        limit = { awow_valid_other_ruler_trigger = yes }
        save_scope_as = awow_other_ruler

        awow_collect_potential_spouses_effect = yes

        clear_saved_scope = awow_other_ruler
    }
}

awow_collect_potential_spouses_effect = {
    if = {
        limit = { scope:awow_single_gal = scope:awow_ruler }
        awow_collect_ruler_potential_spouses_effect = yes
    }
    else = {
        awow_collect_child_potential_spouses_effect = yes
    }
}

awow_collect_ruler_potential_spouses_effect = {
    if = {
        limit = { awow_weighted_potential_spouse_trigger = yes }
        awow_store_tertiary_potential_spouse_effect = yes
    }
    every_child = {
        limit = { awow_weighted_potential_spouse_trigger = yes }
        awow_store_primary_potential_spouse_effect = yes
    }
    every_close_family_member = {
        limit = {
            NOT = { is_child_of = scope:awow_other_ruler }
            awow_weighted_potential_spouse_trigger = yes
        }
        awow_store_secondary_potential_spouse_effect = yes
    }
}

awow_collect_child_potential_spouses_effect = {
    if = {
        limit = {
            is_landed = yes
            awow_weighted_potential_spouse_trigger = yes
        }
        awow_store_primary_potential_spouse_effect = yes
    }
    every_child = {
        limit = { awow_weighted_potential_spouse_trigger = yes }
        awow_store_secondary_potential_spouse_effect = yes
    }
    every_close_family_member = {
        limit = {
            NOT = { is_child_of = scope:awow_other_ruler }
            awow_weighted_potential_spouse_trigger = yes
        }
        awow_store_tertiary_potential_spouse_effect = yes
    }
}

# --------- Priority Weights --------- #
awow_store_primary_potential_spouse_effect = {
    scope:awow_ruler = {
        set_variable = {
            name = awow_potential_spouse_weight
            value = 2000
        }
    }
    awow_apply_other_ruler_relationship_weight_effect = yes
    awow_store_best_potential_spouse_effect = yes
}

awow_store_secondary_potential_spouse_effect = {
    scope:awow_ruler = {
        set_variable = {
            name = awow_potential_spouse_weight
            value = 500
        }
    }
    awow_apply_other_ruler_relationship_weight_effect = yes
    awow_store_best_potential_spouse_effect = yes
}

awow_store_tertiary_potential_spouse_effect = {
    scope:awow_ruler = {
        set_variable = {
            name = awow_potential_spouse_weight
            value = 100
        }
    }
    awow_apply_other_ruler_relationship_weight_effect = yes
    awow_store_best_potential_spouse_effect = yes
}

awow_apply_other_ruler_relationship_weight_effect = {
    awow_apply_potential_spouse_age_gap_weight_effect = yes
    if = {
        limit = {
            scope:awow_other_ruler = { awow_other_ruler_direct_vassal_trigger = yes }
        }
        awow_apply_direct_vassal_tier_weight_effect = yes
    }
    else_if = {
        limit = {
            scope:awow_other_ruler = { awow_other_ruler_fellow_vassal_trigger = yes }
        }
        scope:awow_ruler = {
            change_variable = {
                name = awow_potential_spouse_weight
                multiply = 3
            }
        }
    }
    else_if = {
        limit = {
            scope:awow_other_ruler = { awow_other_ruler_allied_trigger = yes }
        }
        scope:awow_ruler = {
            change_variable = {
                name = awow_potential_spouse_weight
                multiply = 2
            }
        }
    }
    else = {
        awow_apply_foreign_other_ruler_tier_gap_weight_effect = yes
    }
}

# Direct-vassal weight scales by the actual subordinate's highest held title tier: barony x1, county x3, duchy x6, kingdom x9, empire x15.
awow_apply_direct_vassal_tier_weight_effect = {
    scope:awow_ruler = {
        change_variable = {
            name = awow_potential_spouse_weight
            multiply = 3
        }
    }
    if = {
        limit = {
            scope:awow_other_ruler = { is_vassal_of = scope:awow_ruler }
        }
        scope:awow_other_ruler = { save_temporary_scope_as = awow_direct_vassal }
    }
    else_if = {
        limit = {
            scope:awow_ruler = { is_vassal_of = scope:awow_other_ruler }
        }
        scope:awow_ruler = { save_temporary_scope_as = awow_direct_vassal }
    }
    if = {
        limit = { exists = scope:awow_direct_vassal }
        if = {
            limit = { scope:awow_direct_vassal = { highest_held_title_tier = tier_barony } }
            scope:awow_ruler = {
                change_variable = {
                    name = awow_potential_spouse_weight
                    divide = 3
                }
            }
        }
        else_if = {
            limit = { scope:awow_direct_vassal = { highest_held_title_tier >= tier_empire } }
            scope:awow_ruler = {
                change_variable = {
                    name = awow_potential_spouse_weight
                    multiply = 5
                }
            }
        }
        else_if = {
            limit = { scope:awow_direct_vassal = { highest_held_title_tier >= tier_kingdom } }
            scope:awow_ruler = {
                change_variable = {
                    name = awow_potential_spouse_weight
                    multiply = 3
                }
            }
        }
        else_if = {
            limit = { scope:awow_direct_vassal = { highest_held_title_tier >= tier_duchy } }
            scope:awow_ruler = {
                change_variable = {
                    name = awow_potential_spouse_weight
                    multiply = 2
                }
            }
        }
    }
}

awow_apply_potential_spouse_age_gap_weight_effect = {
    while = {
        count = {
            value = scope:awow_single_gal.age
            subtract = age
            abs = yes
            floor = yes
        }
        scope:awow_ruler = {
            change_variable = {
                name = awow_potential_spouse_weight
                multiply = 0.95
            }
        }
    }
}

# Makes a potential spouse from a foreign ruler less likely to be selected if the tier gap is large, but does not prevent it entirely.
awow_apply_foreign_other_ruler_tier_gap_weight_effect = {
    while = {
        count = {
            value = scope:awow_ruler.highest_held_title_tier
            subtract = scope:awow_other_ruler.highest_held_title_tier
            abs = yes
            floor = yes
        }
        scope:awow_ruler = {
            change_variable = {
                name = awow_potential_spouse_weight
                divide = 3
            }
        }
    }
}

awow_store_best_potential_spouse_effect = {
    if = {
        limit = {
            NOT = { is_in_list = awow_weighted_potential_spouses }
        }
        set_variable = {
            name = awow_potential_spouse_weight
            value = scope:awow_ruler.var:awow_potential_spouse_weight
        }
        set_variable = {
            name = awow_potential_spouse_other_ruler
            value = scope:awow_other_ruler
        }
        add_to_list = awow_weighted_potential_spouses
    }
    else_if = {
        limit = {
            scope:awow_ruler.var:awow_potential_spouse_weight > var:awow_potential_spouse_weight
        }
        set_variable = {
            name = awow_potential_spouse_weight
            value = scope:awow_ruler.var:awow_potential_spouse_weight
        }
        set_variable = {
            name = awow_potential_spouse_other_ruler
            value = scope:awow_other_ruler
        }
    }
    scope:awow_ruler = { remove_variable = awow_potential_spouse_weight }
}

# --------- Select Random Spouse --------- #
awow_select_weighted_potential_spouse_effect = {
    random_in_list = {
        list = awow_weighted_potential_spouses
        weight = {
            base = 0
            modifier = {
                add = {
                    add = var:awow_potential_spouse_weight
                }
            }
        }
        save_scope_as = awow_potential_spouse
        var:awow_potential_spouse_other_ruler = {
            save_scope_as = awow_other_ruler
        }
        awow_try_marriage_interaction_effect = yes
        clear_saved_scope = awow_other_ruler
        clear_saved_scope = awow_potential_spouse
    }
    awow_clear_weighted_potential_spouse_pool_effect = yes
}

# --------- Clear Weights --------- #
awow_clear_weighted_potential_spouse_pool_effect = {
    every_in_list = {
        list = awow_weighted_potential_spouses
        remove_variable = awow_potential_spouse_weight
        remove_variable = awow_potential_spouse_other_ruler
        remove_from_list = awow_weighted_potential_spouses
    }
}

# --------- Marriage Interaction --------- #
awow_try_marriage_interaction_effect = {
    if = {
        limit = {
            scope:awow_single_gal = { awow_can_marry_trigger = yes }
            scope:awow_potential_spouse = {
                awow_matchmaking_potential_spouse_trigger = yes
                awow_valid_marriage_interaction_trigger = yes
            }
        }
        run_interaction = {
            interaction = arrange_marriage_interaction
            actor = scope:awow_ruler
            recipient = scope:awow_other_ruler
            secondary_actor = scope:awow_single_gal
            secondary_recipient = scope:awow_potential_spouse
            execute_threshold = accept
        }
    }
}
