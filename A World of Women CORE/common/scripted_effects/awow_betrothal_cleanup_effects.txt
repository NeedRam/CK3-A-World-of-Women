# Clean up a custom same-sex betrothal with no penalties.
# Used when a fringe case (vanilla betrothal/marriage) conflicts
# with our custom betrothal.

awow_cleanup_custom_betrothal_effect = {
    $CHARACTER$ = {
        remove_variable = awow_betrothed_to
        remove_character_flag = awow_same_sex_betrothal
        remove_opinion = {
            target = $BETROTHED$
            modifier = awow_betrothed_opinion
        }
    }
    $BETROTHED$ = {
        remove_variable = awow_betrothed_to
        remove_character_flag = awow_same_sex_betrothal
        remove_opinion = {
            target = $CHARACTER$
            modifier = awow_betrothed_opinion
        }
    }
    # Break alliance between the rulers who arranged the betrothal
    $CHARACTER$ = {
        liege = { save_scope_as = character_liege }
    }
    $BETROTHED$ = {
        liege = { save_scope_as = betrothed_liege }
    }
    scope:character_liege = {
        break_alliance = scope:betrothed_liege
    }
}
