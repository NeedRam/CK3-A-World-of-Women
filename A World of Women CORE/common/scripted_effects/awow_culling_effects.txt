awow_perform_culling = {
    # Reset the running total before we count this pass.
    set_global_variable = {
        name = awow_cull_count
        value = 0
    }

    # Spare the player character (if that rule is enabled).
    if = {
        limit = {
            has_game_rule = awow_exclude_player
        }
        every_player = {
            add_character_flag = awow_excluded_from_culling
        }
    }

    # Spare every player dynasty (if that rule is enabled).
    # This is one world pass: each character is flagged if they share a dynasty
    # with any current player. Do not wrap this in every_player, which would
    # repeat the same global scan once per player.
    if = {
        limit = {
            has_game_rule = awow_exclude_player_dynasty
        }
        every_living_character = {
            limit = {
                is_male = yes
                any_player = {
                    dynasty = prev.dynasty
                }
            }
            add_character_flag = awow_excluded_from_culling
        }
    }


    every_living_character = {
        limit = {
            is_female = yes
            has_character_flag = awow_excluded_from_culling
        }
        remove_character_flag = awow_excluded_from_culling
        remove_variable = awow_culling_protection_owner
    }

    # Cull: every living male that can be culled is killed and counted.
    every_living_character = {
        limit = {
            can_be_culled = yes
        }
        death = natural
        change_global_variable = {
            name = awow_cull_count
            add = 1
        }
    }

    # Reset the cycle counter so the next cull is scheduled from now.
    set_global_variable = {
        name = awow_months_since_cull
        value = 0
    }

    # Debug trace - check Documents/Paradox Interactive/Crusader Kings III/logs for this line.
    debug_log = "AWOW: cull complete"
}
