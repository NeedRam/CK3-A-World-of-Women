# AWOW scripted same sex child birth
awow_born_child_effect = {
    create_character = {
        age = 0
        random_traits = no
        culture = root.culture
        faith = root.faith
        dynasty = inherit
        location = root.location
        mother = root
        gender_female_chance = var:awow_baby_female_chance
        save_scope_as = child
        after_creation = {
            awow_assign_scripted_relations_effect = yes
            awow_assign_baby_to_court_effect = yes
            awow_assign_baby_to_house_effect = yes
            awow_assign_twin_trait_to_baby_effect = yes
            awow_assign_congenital_traits_effect = yes
            awow_trigger_on_birth_on_action_effect = yes
        }
    }

    create_character_memory = {
        type = child_born
        participants = {
            child = scope:child
        }
    }

    clear_saved_scope = child
}

awow_assign_baby_to_court_effect = {
    if = {
        limit = { scope:awow_mother = { is_ruler = yes } }
        scope:awow_mother = { add_courtier = scope:child }
    }
    else_if = {
        limit = {
            exists = scope:awow_other_mother
            scope:awow_other_mother = { is_ruler = yes }
        }
        scope:awow_other_mother = { add_courtier = scope:child }
    }
    else = {
        scope:awow_mother = { add_courtier = scope:child }
    }
}

awow_assign_baby_to_house_effect = {
    if = {
        limit = {
            NOT = { exists = scope:awow_mother.dynasty}
            exists = scope:awow_other_mother.dynasty
        }
        set_house = scope:awow_other_mother.house
    }
}

awow_assign_twin_trait_to_baby_effect = {
    if = {
        limit = { 
            OR = {
                scope:awow_mother = { has_character_flag = awow_pregnancy_will_be_twins_flag }
                scope:awow_mother = { has_character_flag = awow_pregnancy_will_be_triplets_flag }
            }
        }
        add_trait_force_tooltip = twin
    }
}

awow_trigger_on_birth_on_action_effect = {
    scope:awow_mother = { save_scope_as = mother }
    # The engine only accepts a male character in its actual father and
    # real_father parent slots. For the on_action context, however, the
    # co-mother is intentionally supplied as both father targets so every
    # birth-hook consumer sees both parents.
    scope:awow_other_mother = {
        save_scope_as = father
        save_scope_as = real_father
    }
    # on_birth_child branches on this value. AWOW custom births have a known
    # co-mother and should not be routed through vanilla bastard handling.
    save_scope_value_as = { name = is_bastard value = no }

    # Birth integrations expect each parent hook to execute with its parent as
    # root. This also satisfies Carnalitas' requirement to notify both sides.
    scope:awow_mother = {
        trigger_event = { on_action = on_birth_mother }
    }
    scope:awow_other_mother = {
        trigger_event = { on_action = on_birth_father }
    }
    scope:child = { trigger_event = { on_action = on_birth_child } }
    clear_saved_scope = mother
    clear_saved_scope = father
    clear_saved_scope = real_father
    clear_saved_scope = is_bastard
}
