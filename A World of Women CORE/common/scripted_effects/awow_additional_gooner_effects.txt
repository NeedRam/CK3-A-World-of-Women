# Scope: none. Retained for the one-time game-start application so existing
# characters receive their selected world setup immediately.
awow_apply_world_maintenance_effect = {
    every_living_character = {
        awow_apply_world_maintenance_to_character_effect = yes
    }
}

# Scope: character. Called at each character's randomized yearly pulse.
# Vanilla and AWOW both use has_game_rule directly from character scopes.
awow_apply_world_maintenance_to_character_effect = {
    if = {
        limit = { has_game_rule = awow_make_everyone_lustful_enabled }
        awow_assign_horny_trait_effect = { HTRAIT = lustful }
    }
    if = {
        limit = { has_game_rule = awow_make_everyone_fecund_enabled }
        awow_assign_horny_trait_effect = { HTRAIT = fecund }
    }
    if = {
        limit = { awow_using_carn_trigger = yes }
        awow_correct_carn_traits_effect = yes
    }
    if = {
        limit = { has_game_rule = awow_exclude_player_dynasty }
        awow_assign_world_sexualities_player_dynasty_effect = yes
    }
    else_if = {
        limit = { has_game_rule = awow_exclude_player }
        awow_assign_world_sexualities_player_effect = yes
    }
    else = {
        awow_assign_world_sexualities_cull_all_effect = yes
    }
    if = {
        limit = { has_game_rule = awow_maintain_all_ai_character_weight_enabled }
        awow_maintain_weight_effect = yes
    }
}

awow_assign_horny_trait_effect = {
    if = {
        limit = {
            NOT = { has_trait = $HTRAIT$ }
            is_adult = yes
        }
        add_trait = $HTRAIT$
    }
}

# Used when the player and every member of a player dynasty are spared.
awow_assign_world_sexualities_player_dynasty_effect = {
    if = {
        limit = {
            is_male = yes
            OR = {
                is_ai = no
                AND = {
                    exists = dynasty
                    any_player = { dynasty = prev.dynasty }
                }
            }
            NOT = { has_sexuality = heterosexual }
        }
        set_sexuality = heterosexual
    }
    else = {
        awow_assign_world_sexualities_cull_all_effect = yes
    }
}

# Used when the player character alone is spared from the cull.
awow_assign_world_sexualities_player_effect = {
    if = {
        limit = {
            is_ai = no
            is_male = yes
            NOT = { has_sexuality = heterosexual }
        }
        set_sexuality = heterosexual
    }
    else = {
        awow_assign_world_sexualities_cull_all_effect = yes
    }
}

# Every woman is bisexual. The culling setting only changes which males are
# protected from the otherwise universal AI-male homosexual assignment.
awow_assign_world_sexualities_cull_all_effect = {
    if = {
        limit = {
            is_female = yes
            NOT = { has_sexuality = bisexual }
        }
        set_sexuality = bisexual
    }
    else_if = {
        limit = {
            is_ai = yes
            is_male = yes
            NOT = { has_sexuality = homosexual }
        }
        set_sexuality = homosexual
    }
}

awow_maintain_weight_effect = {
    if = {
        limit = {
            is_ai = yes
            is_alive = yes
            OR = {
                current_weight > 10
                current_weight < -10
                target_weight != 0
            }
        }
        # Move both values gradually toward average rather than snapping them.
        if = {
            limit = { current_weight > 10 }
            change_current_weight = -10
        }
        else_if = {
            limit = { current_weight < -10 }
            change_current_weight = 10
        }
        if = {
            limit = { target_weight > 10 }
            change_target_weight = -10
        }
        else_if = {
            limit = { target_weight < -10 }
            change_target_weight = 10
        }
        else_if = {
            limit = { target_weight != 0 }
            # Apply the inverse of the remaining value to land exactly on zero.
            change_target_weight = {
                value = target_weight
                multiply = -1
            }
        }
    }
}
