awow_assign_flat_traits_from_parents_abstracted_effect = {
    if = {
        limit = { awow_both_parents_have_trait_trigger = { TRAIT = $TRAIT$ } }
        random = {
            chance = 90
            add_trait = $TRAIT$
        }
    }
    else_if = {
        limit = { awow_either_parent_has_trait_trigger = { TRAIT = $TRAIT$ } }
        random = {
            chance = 60
            add_trait = $TRAIT$
        }
    }
}

            30 = { add_trait = $GROUP$_good_2 }
            10 = { add_trait = $GROUP$_good_1 }
        }
    }
    else_if = {
        limit = { awow_either_parent_has_trait_trigger = { TRAIT = $GROUP$_good_2 } }
        random_list = {
            15 = { add_trait = $GROUP$_good_3 }
            50 = { add_trait = $GROUP$_good_2 }
            30 = { add_trait = $GROUP$_good_1 }
            5 = { add_trait = $GROUP$_bad_1 }
        }
    }
    else_if = {
        limit = { awow_either_parent_has_trait_trigger = { TRAIT = $GROUP$_good_1 } }
        random_list = {
            15 = { add_trait = $GROUP$_good_2 }
            70 = { add_trait = $GROUP$_good_1 }
            10 = { add_trait = $GROUP$_bad_1 }
            5 = { add_trait = $GROUP$_bad_2 }
        }
    }
    else_if = {
        limit = { awow_either_parent_has_trait_trigger = { TRAIT = $GROUP$_bad_1 } }
        random_list = {
            15 = { add_trait = $GROUP$_good_1 }
            70 = { add_trait = $GROUP$_bad_1 }
            10 = { add_trait = $GROUP$_bad_2 }
            5 = { add_trait = $GROUP$_bad_3 }
        }
    }
    else_if = {
        limit = { awow_either_parent_has_trait_trigger = { TRAIT = $GROUP$_bad_2 } }
        random_list = {
            15 = { add_trait = $GROUP$_good_1 }
            25 = { add_trait = $GROUP$_bad_1 }
            50 = { add_trait = $GROUP$_bad_2 }
            10 = { add_trait = $GROUP$_bad_3 }
        }
    }
    else_if = {
        limit = { awow_either_parent_has_trait_trigger = { TRAIT = $GROUP$_bad_3 } }
        random_list = {
            5 = { add_trait = $GROUP$_good_1 }
            15 = { add_trait = $GROUP$_bad_1 }
            30 = { add_trait = $GROUP$_bad_2 }
            50 = { add_trait = $GROUP$_bad_3 }

awow_upgrade_tiered_traits_vanilla_abstracted_effect = {
    if = {
        limit = { has_trait = $GROUP$_good_2 }
        remove_trait = $GROUP$_good_2
        random_list = {
            10 = { add_trait = $GROUP$_good_3 }
            90 = { add_trait = $GROUP$_good_2 }
        }
    }
    else_if = {
        limit = { has_trait = $GROUP$_good_1 }
        remove_trait = $GROUP$_good_1
        random_list = {
            3 = { add_trait = $GROUP$_good_3 }
            12 = { add_trait = $GROUP$_good_2 }
            85 = { add_trait = $GROUP$_good_1 }
        }
    }
    else_if = {
        limit = { has_trait = $GROUP$_bad_1 }
        remove_trait = $GROUP$_bad_1
        random_list = {
            2 = { add_trait = $GROUP$_good_3 }
            5 = { add_trait = $GROUP$_good_2 }
            23 = { add_trait = $GROUP$_good_1 }
            70 = { add_trait = $GROUP$_bad_1 }
        }
    }
    else_if = {
        limit = { has_trait = $GROUP$_bad_2 }
        remove_trait = $GROUP$_bad_2
        random_list = {
            1 = { add_trait = $GROUP$_good_3 }
            3 = { add_trait = $GROUP$_good_2 }
            11 = { add_trait = $GROUP$_good_1 }
            25 = { add_trait = $GROUP$_bad_1 }
            60 = { add_trait = $GROUP$_bad_2 }
        }
    }
    else_if = {
        limit = { has_trait = $GROUP$_bad_3 }
        remove_trait = $GROUP$_bad_3
        random_list = {
            1 = { add_trait = $GROUP$_good_3 }
            3 = { add_trait = $GROUP$_good_2 }
            6 = { add_trait = $GROUP$_good_1 }
            15 = { add_trait = $GROUP$_bad_1 }
            25 = { add_trait = $GROUP$_bad_2 }
            50 = { add_trait = $GROUP$_bad_3 }
        }
    }
}

awow_remove_all_genetic_traits_effect = {
    if = {
		limit = {
			scope:child = {
				has_trait = pure_blooded
			}
		}
		scope:child = {
			remove_trait = pure_blooded
		}
	}
    awow_remove_all_tiered_genetic_trait_groups_effect = yes
}
