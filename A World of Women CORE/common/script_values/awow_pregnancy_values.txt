awow_impregnation_chance_value = {
    # Both partners are checked immediately before this value is rolled. Average their
    # fertility rather than adding it outright: two 100% fertile women produce
    # a ~29% conception chance on the randomized annual pulse.
    value = scope:awow_mother.fertility
    add = { value = scope:awow_other_mother.fertility }
    # ( ( ( 4.75 * 12 ) / 365 ) * 369 ) / 2
    multiply = 14.406

    # Give a ruler mother without a living child a deterministic heir
    # adjustment. All mothers then receive the same per-child reduction.
    if = {
        limit = {
            scope:awow_mother = {
                is_ruler = yes
                NOT = {
                    any_child = {
                        is_alive = yes
                    }
                }
            }
        }
        add = 100
    }
    every_child = {
        if = {
            limit = { is_alive = yes }
            add = -40
        }
    }
    if = {
        limit = { awow_using_elf_destiny_trigger = yes }
        if = {
            limit = { is_any_real_elf_type = yes }
            add = 21.609 # 75% more children from elves
        }
        else_if = {
            limit = { is_any_elf_type = yes }
            add = 11.525 # 40% more children from half elves
        }
    }
    round = yes
    min = 15
    max = 95
}

awow_pregnancy_twin_chance_value = {
    # Fecund carriers remain more likely to have twins, while especially
    # fertile conceptions receive a small additional chance.
    value = 2
    add = {
        value = awow_impregnation_chance_value
        subtract = 50
        multiply = 0.05
        min = 0
    }
    if = {
        limit = { scope:awow_mother = { has_trait = fecund } }
        add = 3
    }
    if = {
        limit = { scope:awow_mother = { has_trait = infertile } }
        subtract = 5
    }
    round = yes
    min = 0
    max = 95
}

awow_pregnancy_triplet_chance_value = {
    # Preserve wedding-fertility variation at one tenth of the old rates.
    value = 0
    if = {
        limit = {
            OR = {
                scope:awow_mother = { has_relation_wedding_very_good_fertility = scope:awow_other_mother }
                scope:awow_other_mother = { has_relation_wedding_very_good_fertility = scope:awow_mother }
            }
        }
        add = 2
    }
    else_if = {
        limit = {
            OR = {
                scope:awow_mother = { has_relation_wedding_good_fertility = scope:awow_other_mother }
                scope:awow_other_mother = { has_relation_wedding_good_fertility = scope:awow_mother }
            }
        }
        add = 1
    }
    round = yes
    min = 0
    max = 95
}
