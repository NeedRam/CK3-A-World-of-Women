awow_has_beauty_trait_elf_trigger = {
	OR = {
		awow_has_generic_beauty_trait_elf_trigger = yes
		awow_has_exotic_beauty_trait_elf_trigger = yes
		awow_has_unconventional_beauty_trait_elf_trigger = yes
	}
}

awow_either_parent_has_beauty_trait_elf_trigger = {
	OR = {
		scope:awow_mother = { awow_has_beauty_trait_elf_trigger = yes }
		scope:awow_other_mother = { awow_has_beauty_trait_elf_trigger = yes }
	}
}

	}
