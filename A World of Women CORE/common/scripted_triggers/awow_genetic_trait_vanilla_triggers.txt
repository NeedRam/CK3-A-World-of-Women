##################################
# ----------- Beauty ----------- #
##################################

awow_has_beauty_trait_vanilla_trigger = {
	OR = {
		has_trait = beauty_good_3
		has_trait = beauty_good_2
		has_trait = beauty_good_1
		has_trait = beauty_bad_1
		has_trait = beauty_bad_2
		has_trait = beauty_bad_3
	}
}

awow_either_parent_has_beauty_trait_vanilla_trigger = {
	OR = {
		scope:awow_mother = { awow_has_beauty_trait_vanilla_trigger = yes }
		scope:awow_other_mother = { awow_has_beauty_trait_vanilla_trigger = yes }
	}
}

awow_both_parent_have_beauty_trait_vanilla_trigger = {
	AND = {
		scope:awow_mother = { awow_has_beauty_trait_vanilla_trigger = yes }
		scope:awow_other_mother = { awow_has_beauty_trait_vanilla_trigger = yes }
	}
}
