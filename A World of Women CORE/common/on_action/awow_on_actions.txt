on_game_start_after_lobby = {
    on_actions = {
        awow_game_start_setup
    }
}

awow_game_start_setup = {
    effect = {
        # The culling scheduler is a scope = none event chain. Do not start it
        # from a player character: the chain must survive any ruler's death.
        if = {
            limit = { NOT = { has_global_variable = awow_culling_initialized } }
            set_global_variable = {
                name = awow_culling_initialized
                value = yes
            }
            debug_log = "AWOW Culling: Prep START"
            trigger_event = awow_culling.1000
            debug_log = "AWOW Culling: Prep DONE"
        }
        
        every_player = {
            if = {
                limit = { NOT = { has_global_variable = awow_preg_pulse_initialized } }
                set_global_variable = {
                    name = awow_preg_pulse_initialized
                    value = yes
                }
                debug_log = "AWOW SS-PREG: Prep START"
                awow_monthly_pregnancy_tick_effect = yes
                debug_log = "AWOW SS-PREG: Prep DONE"
            }          
        }
        debug_log = "AWOW Assigning Horny Traits START"
        every_living_character = {
            awow_assign_horny_traits_effect = yes
        }
        debug_log = "AWOW Assigning Horny Traits DONE"
        debug_log = "AWOW Assigning Sexualities START"
        every_living_character = {
            awow_assign_world_sexualities_effect = yes
        }
        debug_log = "AWOW Assigning Sexualities DONE"
        debug_log = "AWOW Maintaining Weights START"
        every_living_character = {
            awow_maintain_weight_effect = yes
        }
        debug_log = "AWOW Maintaining Weights DONE"
    }
}

on_pregnancy_mother = {
    events = {
        awow_pregnancy.1000
    }
}

on_birth_child = {
    events = {
        awow_pregnancy.2000
    }
}
