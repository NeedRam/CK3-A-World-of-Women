on_game_start_after_lobby = {
    on_actions = {
        awow_game_start_setup
    }
}

awow_game_start_setup = {
    effect = {
        # The culling scheduler is a scope = none event chain. Do not start it
        # from a player character: the chain must survive any ruler's death.
        if = {
            limit = { NOT = { has_global_variable = awow_culling_initialized } }
            set_global_variable = {
                name = awow_culling_initialized
                value = yes
            }
            debug_log = "AWOW Culling: Prep START"
            trigger_event = awow_culling.1000
            debug_log = "AWOW Culling: Prep DONE"
        }
        
        # Like culling, the pregnancy pulse owns only global state and scans
        # the world itself. Start it from a scope-free event chain rather than
        # binding it to whichever player character happens to be iterated first.
        if = {
            limit = { NOT = { has_global_variable = awow_preg_pulse_initialized } }
            set_global_variable = {
                name = awow_preg_pulse_initialized
                value = yes
            }
            debug_log = "AWOW SS-PREG: Prep START"
            trigger_event = awow_pregnancy.0010
            debug_log = "AWOW SS-PREG: Prep DONE"
        }
        debug_log = "AWOW World Maintenance: START"
        awow_apply_world_maintenance_effect = yes
        debug_log = "AWOW World Maintenance: DONE"
    }
}

on_pregnancy_mother = {
    events = {
        awow_pregnancy.2000
    }
}

on_birth_child = {
    events = {
        awow_pregnancy.1000
    }
}

# Carnalitas invokes this immediately before clearing its custom-pregnancy
# variables. The event filters on AWOW's own pregnancy type.
carn_on_end_custom_pregnancy = {
    events = {
        awow_pregnancy.0040
    }
}
