# Called every Jan 1st
# Root does not exist
yearly_global_pulse = {
    effect = {
        every_living_character = {
            awow_assign_horny_traits_effect = yes
            awow_assign_world_sexualities_effect = yes
            awow_maintain_weight_effect = yes
        }
    }
    on_actions = {
        awow_yearly_ruler_marriage_pulse
        awow_betrothal_fulfillment_pulse
    }
}

awow_yearly_ruler_marriage_pulse = {
    trigger = {
        exists = any_player
    }
    effect = {
        every_living_character = {
            limit = {
                is_ai = yes
                is_adult = yes
                is_playable_character = yes
                awow_can_marry_trigger = yes
            }
            trigger_event = {
                id = awow_ai_marriage.0001
                days = { 1 360 }
            }
        }
    }
}

awow_betrothal_fulfillment_pulse = {
    trigger = {
        exists = any_player
    }
    effect = {
        awow_betrothal_fulfillment_effect = yes
    }
}
