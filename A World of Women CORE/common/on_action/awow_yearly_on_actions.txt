# CK3 invokes this once per character on individually randomized dates. These
# handlers replace the former January 1 global scans.
random_yearly_everyone_pulse = {
    on_actions = {
        awow_random_yearly_ruler_marriage_pulse
        awow_random_yearly_betrothal_maintenance_pulse
        awow_random_yearly_world_maintenance_pulse
    }
}

awow_random_yearly_ruler_marriage_pulse = {
    trigger = {
        exists = any_player
        is_ai = yes
        is_adult = yes
        is_playable_character = yes
        awow_can_marry_trigger = yes
    }
    events = {
        awow_ai_marriage.0001
    }
}

awow_random_yearly_betrothal_maintenance_pulse = {
    trigger = {
        exists = any_player
        OR = {
            has_character_flag = awow_same_sex_betrothal
            has_character_flag = awow_betrothal_processing
        }
    }
    effect = {
        awow_betrothal_maintenance_effect = yes
    }
}

awow_random_yearly_world_maintenance_pulse = {
    effect = {
        awow_apply_world_maintenance_to_character_effect = yes
    }
}
