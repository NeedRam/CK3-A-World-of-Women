namespace = awow_ai_marriage

# Fires from the selected AWOW AI matchmaking scheduler for each eligible AI
# ruler.
# Unmarried adult AI rulers attempt to find spouses for themselves and their
# adult children.
awow_ai_marriage.0001 = {
    type = character_event
    hidden = yes
    trigger = {
        awow_can_marry_trigger = yes
    }
    immediate = {
        awow_ai_ruler_find_spouse_effect = yes
    }
}

awow_ai_marriage.0002 = {
    type = character_event
    hidden = yes
    trigger = {
        any_child = {
            is_adult = yes
            awow_can_marry_trigger = yes
        }
    }
    immediate = {
        awow_ai_ruler_find_spouse_for_child_effect = yes
    }
}
