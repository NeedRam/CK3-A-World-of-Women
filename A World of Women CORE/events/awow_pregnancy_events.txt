namespace = awow_pregnancy

# Per-character pregnancy pulse, invoked at a randomized point each year.
awow_pregnancy.0021 = {
    type = character_event
    hidden = yes

    trigger = {
        awow_same_sex_can_bear_trigger = yes
    }

    immediate = {
        awow_pregnancy_pulse_effect = yes
    }
}

# Revalidate and resolve a same-sex conception attempt immediately after the
# carrying mother and other mother have been selected.
awow_pregnancy.0030 = {
    type = character_event
    hidden = yes

    immediate = {
        if = {
            limit = {
                awow_same_sex_can_start_pregnancy_trigger = yes
                exists = scope:awow_other_mother
                awow_same_sex_can_conceive_with_trigger = {
                    PARTNER = scope:awow_other_mother
                }
            }
            save_scope_as = awow_mother
            random = {
                chance = awow_impregnation_chance_value
                awow_start_same_sex_pregnancy_effect = yes
            }
        }
    }
}
