namespace = awow_pregnancy

# 0001 - Scripted birth for a same-sex (female-female) pregnancy.
# Fired on the pregnant mother 280 days after the AWOW pregnancy starts.
# Supports singleton, twin, and triplet births via character flags set during conception.
awow_pregnancy.0001 = {
    type = character_event
    hidden = yes
    trigger = {
        is_alive = yes
        has_variable = awow_pregnancy_active
    }
    immediate = {

        # debug_log = "AWOW SS-PREG: borning childs"
        awow_baby_female_chance_effect = yes

        awow_born_child_effect = yes
        scope:child = { save_scope_as = awow_first_newborn }
        # debug_log = "AWOW SS-PREG: 1th child borned"

        if = {
            limit = { has_character_flag = awow_pregnancy_will_be_twins_flag }
            awow_born_child_effect = yes
            scope:child = { save_scope_as = awow_second_newborn }
            # debug_log = "AWOW SS-PREG: another child borned"
        }

        if = {
            limit = { has_character_flag = awow_pregnancy_will_be_triplets_flag }
            awow_born_child_effect = yes
            scope:child = { save_scope_as = awow_third_newborn }
            # debug_log = "AWOW SS-PREG: 3st childs born"
        }

        awow_clear_same_sex_pregnancy_state_effect = yes
        carn_end_custom_pregnancy_effect = yes

        # debug_log = "AWOW SS-PREG: removed traits and flags child borned"

        # Keep each newborn available until both player notifications have
        # been sent, so twin and triplet announcements name every child.
        if = {
            limit = { is_ai = no }
            trigger_event = {
                id = awow_pregnancy.0002
            }
        }

        if = {
            limit = {
                exists = scope:awow_other_mother
                scope:awow_other_mother = {
                    is_ai = no
                    is_alive = yes
                }
            }
            scope:awow_other_mother = {
                trigger_event = {
                    id = awow_pregnancy.0003
                }
            }
        }

        clear_saved_scope = child
        clear_saved_scope = awow_first_newborn
        clear_saved_scope = awow_second_newborn
        clear_saved_scope = awow_third_newborn
        clear_saved_scope = awow_mother
        clear_saved_scope = awow_other_mother
        # debug_log = "AWOW SS-PREG: not ai popups sent"
    }
}

# 0002 - Visible birth announcement for a female player mother (carrying mother).
awow_pregnancy.0002 = {
    type = character_event
    hidden = no
    theme = pregnancy

    trigger = {
        is_alive = yes
        is_female = yes
    }

    title = awow_pregnancy_0002_t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:awow_third_newborn }
                desc = awow_pregnancy_0002_triplets_d
            }
            triggered_desc = {
                trigger = { exists = scope:awow_second_newborn }
                desc = awow_pregnancy_0002_twins_d
            }
            desc = awow_pregnancy_0002_d
        }
    }

    option = {
        name = awow_pregnancy_0002_ok
        trigger = { always = yes }
    }
}

# 0030 - Start Pregnancy
awow_pregnancy.0030 = {
    type = character_event
    hidden = yes

    immediate = {
        # The scheduler queues a possible conception, not a successful one.
        # Revalidate here because a normal or custom pregnancy may have begun
        # during the 0-365 day delay.
        if = {
            limit = {
                awow_same_sex_can_start_pregnancy_trigger = yes
                exists = var:awow_pending_other_mother
                # The queued partner may no longer be nearby, available, or
                # related to the carrier when the delayed conception resolves.
                awow_same_sex_can_conceive_with_trigger = {
                    PARTNER = var:awow_pending_other_mother
                }
            }
            save_scope_as = awow_mother
            var:awow_pending_other_mother = {
                save_scope_as = awow_other_mother
            }
            random = {
                chance = awow_impregnation_chance_value
                awow_start_same_sex_pregnancy_effect = yes
            }
        }
        remove_variable = awow_pending_other_mother
        remove_character_flag = awow_same_sex_pregnancy_pending
    }
}

# 0040 - Carnalitas ends a custom pregnancy.
# Carnalitas fires this hook before it removes its pregnancy-type variable, so
# verify that the ending pregnancy is AWOW's before clearing AWOW-only state.
# Any already-queued AWOW birth event then fails its active-state trigger.
awow_pregnancy.0040 = {
    type = character_event
    hidden = yes

    trigger = {
        has_variable = awow_pregnancy_active
        carn_has_custom_pregnancy_type_trigger = {
            PREGNANCY_TYPE = awow_same_sex
        }
    }

    immediate = {
        awow_clear_same_sex_pregnancy_state_effect = yes
    }
}

# Birth gender assignment event
# Fires on_pregnancy, checks game rules to set the unborn child's gender.
awow_pregnancy.2000 = {
    type = character_event
    hidden = yes

    immediate = {
        scope:real_father = { save_scope_as = real_father }
        # Player baby gender
        if = {
            limit = {
                OR = {
                    # Checks if the player is the mother or father
                    is_ai = no
                    scope:real_father = { is_ai = no }
                }
            }
            awow_birth_gender_player_roll = yes
        }
        # Player Dynasty baby gender
        else_if = {
            limit = {
                AND = {
                    # Ensures both parents are AI
                    is_ai = yes
                    scope:real_father = { is_ai = yes }
                }
                OR = {
                    # Mother is in a player dynasty
                    AND = {
                        exists = dynasty
                        any_player = {
                            dynasty = prev.dynasty
                        }
                    }
                    # Father is in a player dynasty
                    scope:real_father = {
                        AND = {
                            exists = dynasty
                            any_player = {
                                dynasty = prev.dynasty
                            }
                        }
                    }
                }
            }
            awow_birth_gender_dynasty_roll = yes
        }
        # Non-player / non-player-dynasty baby gender
        else = {
            awow_birth_gender_world_roll = yes
        }
    }
}
