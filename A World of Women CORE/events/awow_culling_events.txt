namespace = awow_culling

# Startup setup event - runs immediately, sets up variables and schedules first cull
awow_culling.1000 = {
    type = character_event
    hidden = yes

    immediate = {
        # Set initial cull frequency to 6 months
        set_global_variable = {
            name = awow_culling_frequency
            value = 6
        }

        # Initialize months since cull counter
        set_global_variable = {
            name = awow_months_since_cull
            value = 0
        }

        # Initialize cull count so the tooltip always has a value to read
        set_global_variable = {
            name = awow_cull_count
            value = 0
        }

        # Schedule first cull in 30 days
        trigger_event = {
            id = awow_culling.1001
            days = 30
        }
    }
}

# Initial cull - The Vanishing (with narrative popup)
# Fires 30 days after game start
awow_culling.1001 = {
    type = character_event

    title = awow_culling.1001.title
    desc = awow_culling.1001.desc

    left_portrait = {
        character = root
        animation = shock
        camera = camera_event_center_pointing_right
    }
    theme = death
    override_background = activity_witch_ritual
    override_effect_2d = { reference = fog }

    immediate = {
        # Initialize cull count
        set_global_variable = {
            name = awow_cull_count
            value = 0
        }

        # Perform the culling
        awow_perform_culling = yes

        # Mark that initial cull is done
        set_global_variable = {
            name = awow_initial_cull_done
            value = yes
        }

        # Schedule the recurring monthly pulse (only once, from the first player to reach here)
        if = {
            limit = {
                NOT = { has_global_variable = awow_pulse_scheduled }
            }
            set_global_variable = {
                name = awow_pulse_scheduled
                value = yes
            }
            trigger_event = {
                id = awow_culling.1002
                days = 30
            }
        }
    }

    option = {
        name = awow_culling.1001.a
    }
}

# Regular culling event (hidden - replaced by bottom-right interface message)
# The actual culling is performed once, globally, by awow_culling.1002
# before the interface message is shown, so awow_cull_count already holds the total.
awow_culling.0002 = {
    type = character_event
    hidden = yes

    title = awow_culling.0002.title
    desc = awow_culling.0002.desc

    theme = death

    option = {
        name = awow_culling.0002.a
    }
}

# Recurring monthly culling pulse
# Runs every ~30 days, checks if a cull is due, and sends a bottom-right alert instead of a popup.
awow_culling.1002 = {
    type = character_event
    hidden = yes

    immediate = {
        # Advance the cycle counter
        change_global_variable = {
            name = awow_months_since_cull
            add = 1
        }

        # Check if it's time to cull
        if = {
            limit = {
                global_var:awow_months_since_cull >= global_var:awow_culling_frequency
            }
            # Perform the world-wide cull exactly once
            awow_perform_culling = yes

            # Show a single bottom-right alert to each human player
            every_player = {
                send_interface_message = {
                    type = awow_culling_message
                }
            }
        }

        # Schedule the next pulse in ~30 days
        trigger_event = {
            id = awow_culling.1002
            days = 30
        }
    }

    option = {
        name = awow_culling.1002.a
    }
}
