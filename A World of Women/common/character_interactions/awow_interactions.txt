awow_flag_exclude_interaction = {
	category = interaction_category_diplomacy
	desc = awow_flag_exclude_interaction_desc
	common_interaction = yes
	auto_accept = yes


	is_shown = {
		scope:recipient = {
			is_alive = yes
			is_female = no
			NOT = { has_character_flag = awow_excluded_from_culling }
		}
		save_scope_as = awow_culling_flag
	}
	
	is_valid = {
		scope:recipient = {
			is_alive = yes
			is_female = no
			NOT = { has_character_flag = awow_excluded_from_culling }
		}
	}
	
	on_accept = {
		scope:recipient = {
			add_character_flag = awow_excluded_from_culling
		}
		trigger_event = {
			id = awow.0010
			days = 0
		}
	}
}

awow_flag_include_interaction = {
	category = interaction_category_diplomacy
	desc = awow_flag_include_interaction_desc
	common_interaction = yes
	auto_accept = yes

		scope:recipient = {
			is_alive = yes

		}

	is_valid = {
		scope:recipient = {

		}
	}
	
	on_accept = {

			days = 0
		}
	}
}
