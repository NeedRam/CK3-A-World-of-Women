# TEST interaction: force a same-sex pregnancy with the chosen female partner,
# bypassing the fertility roll. Reuses the exact same start logic as the AI path.
# Usable by any female character on a female spouse/concubine. For debugging only.
awow_test_same_sex_pregnancy_interaction = {
    category = interaction_category_uncategorized
    icon = debug_chars
    common_interaction = yes
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes
    auto_accept = yes

    is_shown = {
        scope:recipient = {
            is_alive = yes
            is_female = yes
            OR = {
                is_spouse_of = scope:actor
                is_concubine_of = scope:actor
            }
        }
    }

    is_valid = {
        scope:recipient = {
            is_alive = yes
            is_female = yes
            can_have_children = yes
            age <= 45
            NOT = { is_pregnant = yes }
            carn_has_any_custom_pregnancy_trigger = no
            OR = {
                is_spouse_of = scope:actor
                is_concubine_of = scope:actor
            }
        }
    }

    on_accept = {
        debug_log = "AWOW SS-PREG TEST: the player forced pregnancy on their spouse"
        scope:actor = {
            save_scope_as = father
            save_scope_as = awow_other_mother
        }
        scope:recipient = {
            save_scope_as = mother
            awow_start_same_sex_pregnancy_effect = yes
        }
    }
}
