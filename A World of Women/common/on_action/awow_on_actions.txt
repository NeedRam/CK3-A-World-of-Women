on_game_start_after_lobby = {
    on_actions = {
        awow_game_start_setup
    }
}

awow_game_start_setup = {
    effect = {
        every_player = {
            if = {
                limit = { NOT = { has_global_variable = awow_culling_initalized } }
                set_global_variable = {
                    name = awow_culling_initalized
                    value = yes
                }
                debug_log = "AWOW Culling: Prep START"
                trigger_event = awow.1000
                debug_log = "AWOW Culling: Prep DONE"
            }          
        }
        debug_log = "AWOW BI Girls: Making all girls bi START"
        awow_all_girls_should_be_bi = yes
        debug_log = "AWOW BI Girls: All girls super bi now START"
        every_player = {
            if = {
                limit = { NOT = { has_global_variable = awow_preg_pulse_initalized } }
                set_global_variable = {
                    name = awow_preg_pulse_initalized
                    value = yes
                }
                debug_log = "AWOW SS-PREG: Prep START"
                awow_monthly_pregnancy_tick_effect = yes
                debug_log = "AWOW SS-PREG: Prep DONE"
            }          
        }
        debug_log = "AWOW HOM: Assigning all Historic other Mothers START"
        awow_assign_all_character_history_after_effects = yes
        debug_log = "AWOW HOM: DONE"
        debug_log = "AWOW Backfill: START"
        awow_mass_vanilla_relations_to_scripted_relations_conversion_effect = yes
        debug_log = "AWOW Backfill DONE"
    }
}

on_pregnancy_mother = {
    events = {
        awow.0020
    }
}

on_birth_child = {
    events = {
        awow_pregnancy.1000
    }
}
