# AWOW same-sex (female-female) pregnancy scripted effects
#
# Driven by a monthly AI sweep (see awow_monthly_pregnancy_tick_effect). No male is ever
# involved: a child is produced by create_character with both women as parents.
#
# Visible pregnancy uses the vanilla `pregnant` trait (shows the belly) plus Carnalitas's
# custom-pregnancy variable so `carn_has_any_custom_pregnancy_trigger` / the stage-change
# timer track the pregnancy and clean up correctly.

# ---------------------------------------------------------------------------
# awow_pick_female_chance (requires RULE = player|dynasty|world)
# Reads the active awow_birth_gender_<RULE>_<N> game rule and stores the female
# percentage (0-100) into var:awow_baby_female_chance. The number in the rule name
# IS the female percentage, so each branch simply assigns that value.
# ---------------------------------------------------------------------------
awow_pick_female_chance = { # requires RULE
	if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_100 }
		set_variable = { name = awow_baby_female_chance value = 100 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_99 }
		set_variable = { name = awow_baby_female_chance value = 99 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_95 }
		set_variable = { name = awow_baby_female_chance value = 95 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_90 }
		set_variable = { name = awow_baby_female_chance value = 90 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_80 }
		set_variable = { name = awow_baby_female_chance value = 80 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_75 }
		set_variable = { name = awow_baby_female_chance value = 75 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_67 }
		set_variable = { name = awow_baby_female_chance value = 67 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_50 }
		set_variable = { name = awow_baby_female_chance value = 50 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_33 }
		set_variable = { name = awow_baby_female_chance value = 33 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_25 }
		set_variable = { name = awow_baby_female_chance value = 25 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_20 }
		set_variable = { name = awow_baby_female_chance value = 20 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_10 }
		set_variable = { name = awow_baby_female_chance value = 10 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_5 }
		set_variable = { name = awow_baby_female_chance value = 5 }
	}
	else_if = {
		limit = { has_game_rule = awow_birth_gender_$RULE$_1 }
		set_variable = { name = awow_baby_female_chance value = 1 }
	}
	else = {
		set_variable = { name = awow_baby_female_chance value = 0 }
	}
}

# ---------------------------------------------------------------------------
# awow_baby_female_chance_value
# Produces var:awow_baby_female_chance (0-100) for a scripted birth.
# Must be called with root = the pregnant mother and a saved scope
# `awow_other_mother` = the other woman (her partner). Mirrors the three-tier logic
# in awow.0020 so the two never drift.
# ---------------------------------------------------------------------------
awow_baby_female_chance_value = {
	if = {
		limit = {
			OR = {
				is_ai = no
				scope:awow_other_mother = { is_ai = no }
			}
		}
		awow_pick_female_chance = { RULE = player }
	}
	else_if = {
		limit = {
			AND = {
				is_ai = yes
				scope:awow_other_mother = { is_ai = yes }
				OR = {
					AND = {
						exists = dynasty
						any_player = { dynasty = prev.dynasty }
					}
					scope:awow_other_mother = {
						AND = {
							exists = dynasty
							any_player = { dynasty = prev.dynasty }
						}
					}
				}
			}
		}
		awow_pick_female_chance = { RULE = dynasty }
	}
	else = {
		awow_pick_female_chance = { RULE = world }
	}
}

# ---------------------------------------------------------------------------
# awow_start_same_sex_pregnancy_effect (requires PARTNER_ID)
# Begins the scripted pregnancy on the current root (the pregnant mother).
# ---------------------------------------------------------------------------
awow_start_same_sex_pregnancy_effect = { # requires PARTNER_ID
	debug_log = "AWOW SS-PREG: START [this.GetName] partner_id=$PARTNER_ID$"
	set_variable = { name = awow_other_mother_id value = $PARTNER_ID$ }
	add_trait_force_tooltip = pregnant
	carn_start_custom_pregnancy_effect = {
		PREGNANCY_TYPE = awow_same_sex
		DAYS_UNTIL_NEXT_STAGE = 280
		CAN_BE_ABORTED = no
	}
	debug_log = "AWOW SS-PREG: after_start pregnant=[has_trait=pregnant] carn_var=[exists=var:carn_custom_pregnancy]"
}

# ---------------------------------------------------------------------------
# awow_try_same_sex_pregnancy_effect
# ---------------------------------------------------------------------------
awow_try_same_sex_pregnancy_effect = {
	debug_log = "AWOW SS-PREG: try [this.GetName] ([this.GetID])"
	if = {
		limit = {
			awow_same_sex_can_bear_trigger = yes
			awow_has_ai_female_partner_trigger = yes
		}

		random_spouse = {
			limit = {
				is_female = yes
				is_ai = yes
			}
			save_scope_as = awow_partner
		}
		if = {
			limit = { NOT = { exists = scope:awow_partner } }
			random_consort = {
				limit = {
					is_female = yes
					is_ai = yes
				}
				save_scope_as = awow_partner
			}
		}

		if = {
			limit = { exists = scope:awow_partner }
			debug_log = "AWOW SS-PREG: partner [scope:awow_partner.GetName] ([scope:awow_partner.GetID]) for [this.GetName]"
		}

		if = {
			limit = { character_id < scope:awow_partner.character_id }

			debug_log = "AWOW SS-PREG: decider [this.GetName] fert=[scope:this.fertility] partner_fert=[scope:awow_partner.fertility]"

			set_variable = {
				name = awow_fert_chance
				value = scope:this.fertility
			}
			change_variable = {
				name = awow_fert_chance
				add = scope:awow_partner.fertility
			}
			divide_variable = {
				name = awow_fert_chance
				value = 2
			}
			multiply_variable = {
				name = awow_fert_chance
				value = 20
			}

			debug_log = "AWOW SS-PREG: fert_chance=[var:awow_fert_chance]"

			random = {
				chance = var:awow_fert_chance
				if = {
					limit = { always = yes }
					debug_log = "AWOW SS-PREG: SUCCESS [this.GetName] carries"
					awow_start_same_sex_pregnancy_effect = {
						PARTNER_ID = scope:awow_partner.character_id
					}
				}
				else = {
					debug_log = "AWOW SS-PREG: SUCCESS partner carries"
					scope:awow_partner = {
						awow_start_same_sex_pregnancy_effect = {
							PARTNER_ID = scope:this.character_id
						}
					}
				}
			}
		}
	}
}

# ---------------------------------------------------------------------------
# awow_monthly_pregnancy_tick_effect
# DIAGNOSTIC version: counts AI females and why they don't qualify.
# ---------------------------------------------------------------------------
awow_monthly_pregnancy_tick_effect = {
	debug_log = "AWOW SS-PREG: monthly tick start"

	set_global_variable = { name = awow_diag_total_ai_fem value = 0 }
	set_global_variable = { name = awow_diag_has_partner  value = 0 }
	set_global_variable = { name = awow_diag_spouse_only  value = 0 }
	set_global_variable = { name = awow_diag_can_bear    value = 0 }
	set_global_variable = { name = awow_diag_candidate    value = 0 }

	every_living_character = {
		limit = { is_ai = yes is_female = yes }
		change_global_variable = { name = awow_diag_total_ai_fem add = 1 }
		debug_log = "AWOW SS-PREG: diag_ai_fem"

		if = {
			limit = { awow_has_spouse_partner_trigger = yes }
			change_global_variable = { name = awow_diag_spouse_only add = 1 }
			debug_log = "AWOW SS-PREG: diag_spouse_only"
		}

		if = {
			limit = { awow_has_ai_female_partner_trigger = yes }
			change_global_variable = { name = awow_diag_has_partner add = 1 }
			debug_log = "AWOW SS-PREG: diag_has_partner"
		}

		if = {
			limit = { awow_same_sex_can_bear_trigger = yes }
			change_global_variable = { name = awow_diag_can_bear add = 1 }
			debug_log = "AWOW SS-PREG: diag_can_bear"
		}

		if = {
			limit = {
				awow_same_sex_can_bear_trigger = yes
				awow_has_ai_female_partner_trigger = yes
			}
			change_global_variable = { name = awow_diag_candidate add = 1 }
			debug_log = "AWOW SS-PREG: CANDIDATE FOUND"
			awow_try_same_sex_pregnancy_effect = yes
		}
	}

	debug_log = "AWOW SS-PREG: diag DONE"

	trigger_event = {
		id = awow_pregnancy.0003
		days = 30
	}
}
