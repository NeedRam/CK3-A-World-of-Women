# AWOW autonomous female-female pregnancy system.
# Carnalitas supplies the pregnancy portrait support and fertility script value;
# AWOW owns the pregnancy state and delayed birth event.

awow_start_same_sex_pregnancy_effect = { # requires PARTNER
    set_variable = { name = awow_pregnancy_active value = yes }
    set_variable = { name = awow_other_mother value = $PARTNER$ }
    carn_start_custom_pregnancy_effect = {
        PREGNANCY_TYPE = awow_same_sex
        DAYS_UNTIL_NEXT_STAGE = 280
        CAN_BE_ABORTED = yes
    }
    add_trait_force_tooltip = pregnant
    awow_roll_multiple_children_chance_effect = yes
    trigger_event = {
        id = awow_pregnancy.0001
        days = 280
    }
}

# One annual world scan. Each eligible pair is marked for the year, one woman
# is selected 50/50, then Carnalitas's fertility value performs the conception roll.
awow_try_same_sex_pregnancy_effect = {
    clear_saved_scope = awow_partner
    clear_saved_scope = awow_root
    if = {
        limit = {
            awow_same_sex_can_bear_trigger = yes
            awow_has_ai_female_partner_trigger = yes
        }
        random_spouse = {
            limit = { awow_same_sex_can_bear_trigger = yes }
            save_scope_as = awow_partner
        }
        if = {
            limit = { NOT = { exists = scope:awow_partner } }
            random_consort = {
                limit = { awow_same_sex_can_bear_trigger = yes }
                save_scope_as = awow_partner
            }
        }
        if = {
            limit = { exists = scope:awow_partner }
            save_scope_as = awow_root
            scope:awow_partner = {
                set_variable = {
                    name = awow_same_sex_partner_cooldown
                    value = yes
                    days = 365
                }
            }
            set_variable = {
                name = awow_same_sex_partner_cooldown
                value = yes
                days = 365
            }
            random_list = {
                50 = {
                    random = {
                        chance = awow_impregnation_chance_value
                        save_scope_as = mother
                        scope:awow_partner = { save_scope_as = father }
                        awow_start_same_sex_pregnancy_effect = { PARTNER = scope:awow_partner }
                    }
                }
                50 = {
                    scope:awow_partner = {
                        random = {
                            chance = awow_impregnation_chance_value
                            save_scope_as = father
                            scope:awow_partner = { save_scope_as = mother }
                            awow_start_same_sex_pregnancy_effect = { PARTNER = scope:awow_root }
                        }
                    }
                }
            }
        }
    }
}

awow_roll_multiple_children_chance_effect = {
    if = {
        limit = {
            NOR = {
                has_character_flag = awow_pregnancy_will_be_twins_flag
                has_character_flag = awow_pregnancy_will_be_triplets_flag
            }
        }
        random = {
            chance = awow_pregnancy_twin_chance_value
            add_character_flag = awow_pregnancy_will_be_twins_flag
        }
    }
    if = {
        limit = { NOT = { has_character_flag = awow_pregnancy_will_be_twins_flag } }
        random = {
            chance = awow_pregnancy_triplet_chance_value
            add_character_flag = awow_pregnancy_will_be_triplets_flag
        }
    }
}

awow_monthly_pregnancy_tick_effect = {
    every_living_character = {
        limit = {
            awow_same_sex_can_bear_trigger = yes
            awow_has_ai_female_partner_trigger = yes
        }
        awow_try_same_sex_pregnancy_effect = yes
    }
    trigger_event = {
        id = awow_pregnancy.0003
        days = awow_pregnancy_scheduler_days_value
    }
}
