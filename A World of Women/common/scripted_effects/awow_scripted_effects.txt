awow_perform_culling = {
	# Mark all cullable males
	every_alive_character = {
		limit = { can_be_culled = yes }
		add_character_flag = awow_marked_for_death
	}
	
	# Count and kill them
	every_alive_character = {
		limit = { has_character_flag = awow_marked_for_death }
		kill_character = yes
		change_global_variable = {
			name = awow_cull_count
			add = 1
		}
		remove_character_flag = awow_marked_for_death
	}
	
	# Reset the month counter
	set_global_variable = {
		name = awow_months_since_cull
		value = 0
	}
}
