nivkh_pogibi_2_effect = {
	if = {
		limit = {
			exists = character:nivkh_pogibi_2
			exists = character:nivkh_pogibi_1
		}
		character:nivkh_pogibi_2 = { save_scope_as = awow_child}
		character:nivkh_pogibi_1 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
