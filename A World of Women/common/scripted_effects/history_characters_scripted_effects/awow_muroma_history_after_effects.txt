194160_effect = {
	if = {
		limit = {
			exists = character:194160
			exists = character:194159
		}
		character:194160 = { save_scope_as = awow_child}
		character:194159 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
