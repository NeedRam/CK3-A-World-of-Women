assign_history_emishi_other_mothers = {
	emishi_tsugaru_1_effect = yes
	emishi_tsugaru_2_effect = yes
	emishi_tsugaru_3_effect = yes
	emishi_tsugaru_4_effect = yes
	emishi_tsugaru_5_effect = yes
	emishi_tsugaru_6_effect = yes
	emishi_tsugaru_7_effect = yes
}

emishi_tsugaru_1_effect = {
	if = {
		limit = {
			exists = character:emishi_tsugaru_1
			exists = character:emishi_tsugaru_2
		}
		character:emishi_tsugaru_1 = { save_scope_as = awow_child}
		character:emishi_tsugaru_2 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
