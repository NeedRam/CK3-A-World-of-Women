amis_hoanya_02_fic_effect = {
	if = {
		limit = {
			exists = character:amis_hoanya_02_fic
			exists = character:amis_hoanya_01_fic
		}
		character:amis_hoanya_02_fic = { save_scope_as = awow_child}
		character:amis_hoanya_01_fic = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
