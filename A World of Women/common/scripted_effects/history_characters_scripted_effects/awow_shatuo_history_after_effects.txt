shatuo0002_effect = {
	if = {
		limit = {
			exists = character:shatuo0002
			exists = character:shatuo0001
		}
		character:shatuo0002 = { save_scope_as = awow_child}
		character:shatuo0001 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
