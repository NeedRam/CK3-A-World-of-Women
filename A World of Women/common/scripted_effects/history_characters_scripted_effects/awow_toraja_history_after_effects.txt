assign_history_toraja_other_mothers = {
	toraja_banawa_11_fic_effect = yes
	toraja_wenang_02_fic_effect = yes
	toraja_tamboeko_02_fic_effect = yes
}

toraja_banawa_11_fic_effect = {
	if = {
		limit = {
			exists = character:toraja_banawa_11_fic
			exists = character:toraja_banawa_10_fic
		}
		character:toraja_banawa_11_fic = { save_scope_as = awow_child}
		character:toraja_banawa_10_fic = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
