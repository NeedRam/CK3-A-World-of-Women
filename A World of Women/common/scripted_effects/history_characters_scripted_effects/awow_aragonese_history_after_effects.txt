aragonese0001_effect = {
	if = {
		limit = {
			exists = character:aragonese0001
			exists = character:aragonese0002
		}
		character:aragonese0001 = { save_scope_as = awow_child}
		character:aragonese0002 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
