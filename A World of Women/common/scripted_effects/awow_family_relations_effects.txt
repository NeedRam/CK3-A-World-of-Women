awow_set_parent_effect = {
    if = {
        limit = { NOT = { any_relation = { type = awow_parent } } }
        set_relation_awow_parent =  scope:awow_parent
    }
}

awow_set_child_effect = {
    if = {
        limit = { NOT = { any_relation = { type = awow_child } } }
        set_relation_awow_child = scope:awow_child
    }
}

awow_set_my_sibling_effect = {
    if = {
        limit = { NOT = { any_relation = { type = awow_sibling } } }
        set_relation_awow_sibling = scope:awow_sibling
    }
}

awow_set_sibling_of_effect = {
    if = {
        limit = { NOT = { any_relation = { type = awow_sibling } } }
        set_relation_awow_sibling = scope:awow_child
    }
}

awow_set_grandparent_effect = {
    if = {
        limit = { NOT = { any_relation = { type = awow_grandparent } } }
        set_relation_awow_grandparent = scope:awow_child
    }
}

awow_set_grandchild_effect = {
    if = {
        limit = { NOT = { any_relation = { type = awow_grandchild } } }
        set_relation_awow_grandchild = scope:awow_grandparent
    }
}

awow_assign_scripted_parent_relations_effect = {
    save_scope_as = awow_child
    if = {
        limit = { exists = mother }
        mother = { save_scope_as = awow_parent }
        scope:awow_parent = {
            if = {
                limit = { is_alive = yes }
                awow_set_child_effect = yes
                awow_assign_siblings_effect = yes
                awow_assign_grandparent_relations_effect = yes
            }

        }
        awow_set_parent_effect = yes
        clear_saved_scope = awow_parent
    }
    if = {
        limit = { exists = father }
        father = { save_scope_as = awow_parent }
        scope:awow_parent = {
            if = {
                limit = { is_alive = yes }
                awow_set_child_effect = yes
                awow_assign_siblings_effect = yes
                awow_assign_grandparent_relations_effect = yes
            }
        }
        awow_set_parent_effect = yes
        clear_saved_scope = awow_parent
    }
    if = {
        limit = { exists = scope:awow_other_mother }
        scope:awow_other_mother = {
            if = {
                limit = { is_alive = yes }
                save_scope_as = awow_parent
                awow_set_child_effect = yes
                awow_assign_siblings_effect = yes
                awow_assign_grandparent_relations_effect = yes
            }
        }
        awow_set_parent_effect = yes
        clear_saved_scope = awow_parent
        clear_saved_scope = awow_other_mother
    }
    clear_saved_scope = awow_child
}

awow_assign_siblings_effect = {
    every_child = {
        save_scope_as = awow_sibling
        if = {
            limit = {
                NOT = { this = scope:awow_child }
                is_alive = yes
            }
            awow_set_sibling_of_effect = yes
            scope:awow_child = {
                awow_set_my_sibling_effect = yes
            }
            clear_saved_scope = awow_sibling
        }
    }
    every_relation = {
        type = awow_child
        save_scope_as = awow_sibling
        if = {
            limit = {
                NOT = { this = scope:awow_child }
                is_alive = yes
            }
            awow_set_sibling_of_effect = yes
            scope:awow_child = {
                awow_set_my_sibling_effect = yes
            }
        }
        clear_saved_scope = awow_sibling
    }
}

awow_assign_grandparent_relations_effect = {
    if = {
        limit = { exists = father }
        father = { save_scope_as = awow_grandparent }
        scope:awow_grandparent = {
            if = {
                limit = { is_alive = yes }
                awow_set_grandparent_effect = yes
                scope:awow_child = {
                    awow_set_grandchild_effect = yes
                }
            }
        }
        clear_saved_scope = awow_grandparent
    }
    if = {
        limit = { exists = mother }
        mother = { save_scope_as = awow_grandparent }
        scope:awow_grandparent = {
            if = {
                limit = { is_alive = yes }
                awow_set_grandparent_effect = yes
                scope:awow_child = {
                    awow_set_grandchild_effect = yes
                }
            }

        }
        clear_saved_scope = awow_grandparent
    }
    every_relation = {
        type = awow_parent
        limit = { NOT = { scope:awow_child = { any_relation = { type = awow_grandparent } } } }
        save_scope_as = awow_grandparent
        if = {
            limit = { is_alive = yes }
            awow_set_grandparent_effect = yes
            scope:awow_child = {
                awow_set_grandchild_effect = yes
            }
        }
        clear_saved_scope = awow_grandparent
    }
}


# ---------- birth hook ----------

awow_assign_scripted_birth_relations_effect = {
    awow_assign_scripted_parent_relations_effect = yes
}

# ---------- game-start backfill ----------

awow_backfill_relations_effect = {
    every_living_character = {
        limit = { NOT = { has_character_flag = awow_relations_backfilled } }
        awow_assign_scripted_birth_relations_effect = yes
        add_character_flag = awow_relations_backfilled
    }
    every_living_character = {
        limit = { NOT = { has_character_flag = awow_relations_frontfilled } }
        awow_backfill_2nd_pass_effect = yes
        add_character_flag = awow_relations_frontfilled
    }
}

awow_backfill_2nd_pass_effect = {
    root = { save_scope_as = awow_child }
    every_relation = {
        type = awow_parent
        save_scope_as = awow_parent
        awow_set_parent_effect = yes
        awow_assign_siblings_effect = yes
        awow_assign_grandparent_relations_effect = yes
        scope:awow_child = {
            awow_set_child_effect = yes
        }
    }
    clear_saved_scope = awow_parent
    clear_saved_scope = awow_child
}
