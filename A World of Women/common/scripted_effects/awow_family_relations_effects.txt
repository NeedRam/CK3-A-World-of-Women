# ---------- Parent<>Child ---------- #

awow_set_parent_effect = {
    if = {
        limit = {
            is_alive = yes
            NOT = { any_relation = { type = awow_child } }
            NOT = { any_relation = { type = awow_parent } }
            scope:awow_parent = { is_alive = yes }
        }
        set_relation_awow_parent = scope:awow_parent
        # debug_log = "AWOW RELATIONS: Parent Relation Set"
    }
}

awow_set_child_effect = {
    if = {
        limit = {
            is_alive = yes
            NOT = { any_relation = { type = awow_parent } }
            NOT = { any_relation = { type = awow_child } }
            scope:awow_child = { is_alive = yes }
        }
        set_relation_awow_child = scope:awow_child
        # debug_log = "AWOW RELATIONS: Child Relation Set"
    }
}

# -------------- Birth Hook -------------- #
############################################

awow_convert_vanilla_relations_to_scripted_effect = {
    if = {
        limit = { is_alive = yes }
        save_scope_as = awow_child
        if = {
            limit = {
                exists = scope:awow_child.father
                scope:awow_child.father = { is_alive = yes }
            }
            awow_assign_father_relations_effect = yes
        }
        awow_assign_mother_relations_effect = yes
    }
    clear_saved_scope = awow_child
}

awow_assign_scripted_relations_effect = {
    if = {
        limit = { is_alive = yes }
        save_scope_as = awow_child
        awow_assign_mother_relations_effect = yes
        awow_assign_other_mother_relations_effect = yes
    }
    clear_saved_scope = awow_child
}

# ---------- PARENTS ---------- #

awow_assign_father_relations_effect = {
    # debug_log = "AWOW RELATIONS: in child"
    scope:awow_child.father = { save_scope_as = awow_parent }
    scope:awow_parent = {
        # debug_log = "AWOW RELATIONS: in father"
        awow_set_child_effect = yes
        awow_assign_siblings_effect = yes
        awow_assign_grandparent_relations_effect = yes
    }
    if = {
        limit { scope:awow_parent = { is_alive = yes } }
        awow_set_parent_effect = yes
    }
    clear_saved_scope = awow_parent
}

awow_assign_mother_relations_effect = {
    # debug_log = "AWOW RELATIONS: in child"
    if = {
        if = {
            limit = {
                exists = scope:awow_child.mother
                scope:awow_child.mother = { is_alive = yes }
            }
            scope:awow_child.mother = { save_scope_as = awow_parent }
            scope:awow_parent = {
                if = {
                    # debug_log = "AWOW RELATIONS: in mother"
                    awow_set_child_effect = yes
                    awow_assign_siblings_effect = yes
                    awow_assign_grandparent_relations_effect = yes
                }
            }
            if = {
                limit { scope:awow_parent = { is_alive = yes } }
                awow_set_parent_effect = yes
            }
            clear_saved_scope = awow_parent
        }

    }
}

awow_assign_other_mother_relations_effect = {
    # debug_log = "AWOW RELATIONS: in child"
    scope:awow_other_mother = {
        if = {
            limit = { is_alive = yes }
            save_scope_as = awow_parent
            # debug_log = "AWOW RELATIONS: in other_mother"
            awow_set_child_effect = yes
            awow_assign_siblings_effect = yes
            awow_assign_grandparent_relations_effect = yes
        }
    }
    if = {
        limit { scope:awow_parent = { is_alive = yes } }
        awow_set_parent_effect = yes
    }
    clear_saved_scope = awow_parent
}

# ---------- Siblings ---------- #

awow_assign_siblings_effect = {
    # debug_log = "AWOW RELATIONS: Every Child"
    every_child = {
        if = {
            limit = {
                NOT = { this = scope:awow_child }
                is_alive = yes
                this.mother = scope:awow_parent
            }
            save_scope_as = awow_sibling
            # debug_log = "AWOW RELATIONS: in siblingof"
            awow_set_sibling_of_effect = yes
            scope:awow_child = {
                # debug_log = "AWOW RELATIONS: in mysibling"
                awow_set_my_sibling_effect = yes
            }
        }
        clear_saved_scope = awow_sibling
    }
    # debug_log = "AWOW RELATIONS: Every Relation"
    every_relation = {
        type = awow_child
        if = {
            limit = {
                NOT = { this = scope:awow_child }
                is_alive = yes
                this.mother = scope:awow_parent
            }
            save_scope_as = awow_sibling
            # debug_log = "AWOW RELATIONS: in siblingof"
            awow_set_sibling_of_effect = yes
            scope:awow_child = {
                # debug_log = "AWOW RELATIONS: in mysibling"
                awow_set_my_sibling_effect = yes
            }
        }
        clear_saved_scope = awow_sibling
    }
}

# ---------- Grandparents ---------- #

awow_assign_grandparent_relations_effect = {
    if = {
        limit = { exists = scope:awow_parent.mother }
        scope:awow_parent.mother = { save_scope_as = awow_grandparent }
        scope:awow_grandparent = {
            if = {
                # debug_log = "AWOW RELATIONS: in grandparent"
                awow_set_grandchild_effect = yes
                scope:awow_child = {
                    # debug_log = "AWOW RELATIONS: in grandchild"
                    awow_set_grandparent_effect = yes
                }
            }
        }
        clear_saved_scope = awow_grandparent
    }
    if = {
        limit = { exists = scope:awow_parent.father }
        scope:awow_parent.father = { save_scope_as = awow_grandparent }
        scope:awow_grandparent = {
            if = {
                # debug_log = "AWOW RELATIONS: in grandparent"
                awow_set_grandchild_effect = yes
                scope:awow_child = {
                    # debug_log = "AWOW RELATIONS: in grandchild"
                    awow_set_grandparent_effect = yes
                }
            }
        }
        clear_saved_scope = awow_grandparent
    }
    every_relation = {
        type = awow_parent
        limit = { NOT = { scope:awow_child = { any_relation = { type = awow_grandparent } } } }
        save_scope_as = awow_grandparent
        if = {
            # debug_log = "AWOW RELATIONS: in grandparent"
            awow_set_grandchild_effect = yes
            scope:awow_child = {
                # debug_log = "AWOW RELATIONS: in grandchild"
                awow_set_grandparent_effect = yes
            }
        }
        clear_saved_scope = awow_grandparent
    }
}

awow_mass_vanilla_relations_to_scripted_relations_conversion_effect = {
    every_living_character = {
        limit = { NOT = { has_character_flag = awow_relations_backfilled } }
        awow_convert_vanilla_relations_to_scripted_effect = yes
        add_character_flag = awow_relations_backfilled
    }
    debug_log = "AWOW RELATIONS: Starting BackBACKFILL"
    every_living_character = {
        limit = { NOT = { has_character_flag = awow_relations_frontfilled } }
        awow_backfill_2nd_pass_effect = yes
        add_character_flag = awow_relations_frontfilled
    }
}

awow_backfill_2nd_pass_effect = {
    save_scope_as = awow_child
    every_relation = {
        type = awow_parent
        save_scope_as = awow_parent
        if = {
            awow_assign_siblings_effect = yes
            awow_assign_grandparent_relations_effect = yes
        }
    }
}

assign_historic_father_as_other_mother = {
    save_scope_as = awow_child
    scope:$daddy$ = { save_scope_as = awow_other_mother }
    awow_assign_other_mother_relations_effect = yes
    clear_saved_scope = awow_child
    clear_saved_scope = awow_other_mother
}
