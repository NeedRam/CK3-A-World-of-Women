awow_pregnancy_scheduler_days_value = {
    value = 365
}

awow_impregnation_bearer_value = {
    value = 0
    add = {
        value = scope:mother.fertility
        multiply = 50
    }
    if = {
        limit = { scope:mother = { has_trait = fecund } }
        add = 50
    }
    if = {
        limit = { scope:mother = { has_trait = infertile } }
        subtract = 50
    }
    if = {
        limit = { scope:mother = { has_trait = impotent } }
        subtract = 50
    }
    if = {
        limit = { scope:mother = { effective_age > 45 } }
        subtract = {
            value = scope:mother.effective_age
            subtract = 45
            multiply = 2
            min = 0
        }
    }
    min = 0
    max = 100
}

awow_impregnation_partner_value = {
    value = 0
    add = {
        value = scope:father.fertility
        multiply = 50
    }
    if = {
        limit = { scope:father = { has_trait = fecund } }
        add = 50
    }
    if = {
        limit = { scope:father = { has_trait = infertile } }
        subtract = 50
    }
    if = {
        limit = { scope:father = { has_trait = impotent } }
        subtract = 50
    }
    min = 0
    max = 100
}

awow_impregnation_chance_value = {
    value = 0
    add = awow_impregnation_bearer_value
    add = awow_impregnation_partner_value
    divide = 2
    if = {
        limit = {
            OR = {
                scope:mother = { has_relation_wedding_good_fertility = scope:father }
                scope:father = { has_relation_wedding_good_fertility = scope:mother }
            }
        }
        multiply = 1.25
    }
    else_if = {
        limit = {
            OR = {
                scope:mother = { has_relation_wedding_very_good_fertility = scope:father }
                scope:father = { has_relation_wedding_very_good_fertility = scope:mother }
            }
        }
        multiply = 1.50
    }
    else_if = {
        limit = {
            OR = {
                scope:mother = { has_relation_wedding_bad_fertility = scope:father }
                scope:father = { has_relation_wedding_bad_fertility = scope:mother }
            }
        }
        multiply = 0.50
    }
    else_if = {
        limit = {
            OR = {
                scope:mother = { has_relation_wedding_very_bad_fertility = scope:father }
                scope:father = { has_relation_wedding_very_bad_fertility = scope:mother }
            }
        }
        multiply = 0
    }
    else_if = {
        limit = {
            OR = {
                scope:mother = { is_spouse_of = scope:father }
                scope:father = { is_spouse_of = scope:mother }
            }
        }
        multiply = define:NChildbirth|PRIMARY_SPOUSE_FERTILITY_MULTIPLIER
    }
    else_if = {
        limit = {
            OR = {
                scope:mother = { is_concubine_of = scope:father }
                scope:father = { is_concubine_of = scope:mother }
            }
        }
        multiply = define:NChildbirth|CONCUBINE_FERTILITY_MULTIPLIER
    }
    if = {
        limit = {
            scope:mother = { is_ruler = no }
            scope:father = { is_ruler = no }
        }
        multiply = define:NChildbirth|NON_RULER_FERTILITY_MULTIPLIER
    }
    min = 0
    max = 100
}

# Reads awow_birth_gender_<RULE>_<N> and stores the selected female percentage.
awow_pick_female_chance_value = { # requires RULE = player|dynasty|world
    if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_100 }
        set_variable = { name = awow_baby_female_chance value = 100 }
    }
    else_if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_99 }
        set_variable = { name = awow_baby_female_chance value = 99 }
    }
    else_if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_95 }
        set_variable = { name = awow_baby_female_chance value = 95 }
    }
    else_if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_90 }
        set_variable = { name = awow_baby_female_chance value = 90 }
    }
    else_if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_80 }
        set_variable = { name = awow_baby_female_chance value = 80 }
    }
    else_if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_75 }
        set_variable = { name = awow_baby_female_chance value = 75 }
    }
    else_if = {

        set_variable = { name = awow_baby_female_chance value = 67 }
    }
    else_if = {

    }
    else_if = {

        set_variable = { name = awow_baby_female_chance value = 33 }
    }
    else_if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_25 }
        set_variable = { name = awow_baby_female_chance value = 25 }
    }
    else_if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_20 }
        set_variable = { name = awow_baby_female_chance value = 20 }
    }
    else_if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_10 }
        set_variable = { name = awow_baby_female_chance value = 10 }
    }
    else_if = {
        limit = { has_game_rule = awow_birth_gender_$RULE$_5 }
        set_variable = { name = awow_baby_female_chance value = 5 }

    }
    else = {
        set_variable = { name = awow_baby_female_chance value = 0 }
    }
}

awow_baby_female_chance_value = {
    if = {
        limit = {
            OR = {
                is_ai = no
                scope:awow_other_mother = { is_ai = no }
            }
        }
        value = awow_pick_female_chance = { RULE = player }
    }
    else_if = {
        limit = {
            AND = {
                is_ai = yes
                scope:awow_other_mother = { is_ai = yes }
                OR = {
                    AND = {
                        exists = dynasty
                        any_player = { dynasty = prev.dynasty }
                    }
                    scope:awow_other_mother = {
                        AND = {
                            exists = dynasty
                            any_player = { dynasty = prev.dynasty }
                        }
                    }
                }
            }
        }
        value = awow_pick_female_chance = { RULE = dynasty }
    }
    else = {
        value = awow_pick_female_chance = { RULE = world }
    }
}
