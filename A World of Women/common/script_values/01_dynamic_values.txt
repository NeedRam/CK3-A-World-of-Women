    value = 0
    faith = {
        if = {
            limit = { has_dominant_ruling_gender = dummy_female } # if faith allows females to rule, then add 50% chance
            add = 100
        }
        if = {
            limit = { NOT = { has_dominant_ruling_gender = dummy_male } } # if faith forbids males to rule, then add 50% chance
            add = 100
        }
    }
}
