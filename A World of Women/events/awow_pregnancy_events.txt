namespace = awow_pregnancy

# 0001 - Scripted birth for a same-sex (female-female) pregnancy.
# Fired on the pregnant mother 280 days after the AWOW pregnancy starts.
# Supports singleton, twin, and triplet births via character flags set during conception.
awow_pregnancy.0001 = {
    type = character_event
    hidden = yes
    trigger = {
        is_alive = yes
        has_variable = awow_pregnancy_active
    }
    immediate = {

        create_character = {
            age = 0
            random_traits = no
            culture = root.culture
            faith = root.faith
            dynasty = inherit
            location = root.location
            mother = root
            gender_female_chance = var:awow_baby_female_chance
            save_scope_as = child
        }

        scope:child = {
            awow_assign_scripted_relations_effect = yes
        }

        create_character_memory = {
            type = child_born
            participants = {
                child = scope:child
            }
        }

        if = {
            limit = { has_character_flag = awow_pregnancy_will_be_twins_flag }
            create_character = {
                age = 0
                random_traits = no
                culture = root.culture
                faith = root.faith
                dynasty = inherit
                location = root.location
                mother = root
                gender_female_chance = var:awow_baby_female_chance
                save_scope_as = childsr
            }

            scope:childsr = {
                awow_assign_scripted_relations_effect = yes
            }

            create_character_memory = {
                type = child_born
                participants = {
                    child = scope:childsr
                }
            }
        }

        if = {
            limit = { has_character_flag = awow_pregnancy_will_be_triplets_flag }
            create_character = {
                age = 0
                random_traits = no
                culture = root.culture
                faith = root.faith
                dynasty = inherit
                location = root.location
                mother = root
                gender_female_chance = var:awow_baby_female_chance
                save_scope_as = childssr
            }

            scope:childssr = {
                awow_assign_scripted_relations_effect = yes
            }

            create_character_memory = {
                type = child_born
                participants = {
                    child = scope:childssr
                }
            }
        }

        remove_trait = pregnant
        remove_variable = awow_pregnancy_active
        remove_character_flag = awow_pregnancy_will_be_twins_flag
        remove_character_flag = awow_pregnancy_will_be_triplets_flag
        carn_end_custom_pregnancy_effect = yes


        ######### REDO EVENT POPUPS #########
        if = {
            limit = { is_ai = no }
            scope:child = {
                trigger_event = {
                    id = awow_pregnancy.0002
                }
            }
        }

        if = {
            limit = {
                exists = scope:awow_other_mother
                scope:awow_other_mother = {
                    is_ai = no
                    is_alive = yes
                }
            }
            scope:awow_other_mother = {
                trigger_event = {
                    id = awow_pregnancy.0004
                }
            }
        }
        clear_saved_scope = awow_other_mother
    }
}

# 0002 - Visible birth announcement for a female player mother (carrying mother).
awow_pregnancy.0002 = {
    type = character_event
    hidden = no
    theme = pregnancy

    trigger = {
        is_alive = yes
        is_female = yes
    }

    title = awow_pregnancy_0002_t
    desc = awow_pregnancy_0002_d

    option = {
        name = awow_pregnancy_0002_ok
        trigger = { always = yes }
    }
}

# 0003 - Low-frequency scheduler pulse. Iterates all living women once per year and
# reschedules itself. The pregnancy itself still lasts 280 days.
awow_pregnancy.0003 = {
    type = character_event
    hidden = yes

    immediate = {
        awow_monthly_pregnancy_tick_effect = yes
    }
}

# 0004 - Visible birth announcement for the non-carrying player mother (co-mother).
# Fires only if the non-carrying mother is alive and is the player character.
awow_pregnancy.0004 = {
    type = character_event
    hidden = no
    theme = pregnancy

    trigger = {
        is_alive = yes
        is_female = yes
    }

    title = awow_pregnancy_0004_t
    desc = awow_pregnancy_0004_d

    option = {
        name = awow_pregnancy_0004_ok
        trigger = { always = yes }
    }
}

# 1000 - Assign AWOW family relations for a vanilla birth
awow_pregnancy.1000 = {
    type = character_event
    hidden = yes

    trigger = {
        is_alive = yes
    }

    immediate = {
        awow_convert_vanilla_relations_to_scripted_effect = yes
    }
}
