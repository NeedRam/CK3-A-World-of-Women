namespace = awow_pregnancy

# 0001 - Scripted birth for a same-sex (female-female) pregnancy.
# Fired on the pregnant mother 280 days after the AWOW pregnancy starts.
awow_pregnancy.0001 = {
    type = character_event
    hidden = yes

    trigger = {
        is_alive = yes
        has_variable = awow_pregnancy_active
    }

    immediate = {
        debug_log = "AWOW SS-PREG: BIRTH event fired"
        debug_log = "AWOW SS-PREG: BIRTH checks passed"
        # Resolve the other mother (may have died during the pregnancy).
        var:awow_other_mother = {
            save_scope_as = awow_other_mother
        }

        # Reuse AWOW's configured player/dynasty/world gender rules.
        awow_baby_female_chance_value = yes

        debug_log = "AWOW SS-PREG: BIRTH other_mother resolved"

        # Birth: both women are parents. Female father is data-only and safe.
        if = {
            limit = { exists = scope:awow_other_mother }
            create_character = {
                age = 0
                random_traits = no
                culture = root.culture
                faith = root.faith
                dynasty = inherit
                location = root.location
                mother = root
                gender_female_chance = var:awow_baby_female_chance
                save_scope_as = child
            }
            debug_log = "AWOW SS-PREG: BIRTH created child with both mothers"
        }
        else = {
            create_character = {
                age = 0
                random_traits = no
                culture = root.culture
                faith = root.faith
                dynasty = inherit
                location = root.location
                mother = root
                gender_female_chance = var:awow_baby_female_chance
                save_scope_as = child
            }
            debug_log = "AWOW SS-PREG: BIRTH created child mother only (other dead)"
        }

        # CK3 rejects a female character in the biological father slot. Adoption
        # is the engine-supported way to attach the second woman as a parent.
        if = {
            limit = { exists = scope:awow_other_mother }
            scope:awow_other_mother = { adopt = scope:child }
        }

        # Birth memory so the event shows up in the character's timeline.
        create_character_memory = {
            type = child_born
            participants = {
                child = scope:child
            }
        }

        # Cleanup: remove the visible trait and Carnalitas's custom-pregnancy state.
        remove_trait = pregnant
        remove_variable = awow_pregnancy_active
        remove_variable = awow_other_mother
        carn_end_custom_pregnancy_effect = yes

        # Announce to the player only if the mother is the player's character
        # (rare: a female player). AI mothers get no popup.
        if = {
            limit = { is_ai = no }
            scope:child = {
                trigger_event = {
                    id = awow_pregnancy.0002
                }
            }
        }
    }
}

# 0002 - Visible birth announcement for a female player mother.
awow_pregnancy.0002 = {
    type = character_event
    hidden = no

    trigger = {
        is_alive = yes
        is_female = yes
    }

    title = awow_pregnancy_0002_t
    desc = awow_pregnancy_0002_d

    option = {
        name = awow_pregnancy_0002_ok
        trigger = { always = yes }
    }
}

# 0003 - Low-frequency scheduler pulse. Iterates all living AI women once per year and
# reschedules itself. The pregnancy itself still lasts 280 days.
awow_pregnancy.0003 = {
    type = character_event
    hidden = yes

    immediate = {
        awow_monthly_pregnancy_tick_effect = yes
    }
}
