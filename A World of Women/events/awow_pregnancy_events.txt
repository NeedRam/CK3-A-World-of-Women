namespace = awow_pregnancy

# 0001 - Scripted birth for a same-sex (female-female) pregnancy.
# Fired on the pregnant mother 280 days after the AWOW pregnancy starts.
# Supports singleton, twin, and triplet births via character flags set during conception.
awow_pregnancy.0001 = {
    type = character_event
    hidden = yes
    trigger = {
        is_alive = yes
        has_variable = awow_pregnancy_active
    }
    immediate = {
        # debug_log = "AWOW SS-PREG: borning child"
        awow_baby_female_chance_effect = yes
            create_character = {
                age = 0
                random_traits = no
                culture = root.culture
                faith = root.faith
                dynasty = inherit
                location = root.location
                mother = root
                gender_female_chance = var:awow_baby_female_chance
                save_scope_as = child
                after_creation = {
                    if = {
                        limit = { scope:awow_mother = { is_ruler = yes } }
                        scope:awow_mother = { add_courtier = scope:child }
                    }
                    else_if = {
                        limit = {
                            exists = scope:awow_other_mother
                            scope:awow_other_mother = { is_ruler = yes }
                        }
                        scope:awow_other_mother = { add_courtier = scope:child }
                    }
                    else = {
                        scope:awow_mother = { add_courtier = scope:child }
                    }

                    scope:child = {
                        awow_assign_scripted_relations_effect = yes
                    }

                    if = {
                        limit = {
                            NOT = { exists = scope:awow_mother.dynasty}
                            exists = scope:awow_other_mother.dynasty
                        }
                        set_house = scope:awow_other_mother.house
                    }

                }
            }

        create_character_memory = {
            type = child_born
            participants = {
                child = scope:child
            }
        }

        # debug_log = "AWOW SS-PREG: child borned"

        if = {
            limit = { has_character_flag = awow_pregnancy_will_be_twins_flag }
            create_character = {
                age = 0
                random_traits = no
                culture = root.culture
                faith = root.faith
                dynasty = inherit
                location = root.location
                mother = root
                gender_female_chance = var:awow_baby_female_chance
                save_scope_as = childsr
                after_creation = {
                    if = {
                        limit = { scope:awow_mother = { is_ruler = yes } }
                        scope:awow_mother = { add_courtier = scope:childsr }
                    }
                    else_if = {
                        limit = {
                            exists = scope:awow_other_mother
                            scope:awow_other_mother = { is_ruler = yes }
                        }
                        scope:awow_other_mother = { add_courtier = scope:childsr }
                    }
                    else = {
                        scope:awow_mother = { add_courtier = scope:childsr }
                    }

                    scope:childsr = {
                        awow_assign_scripted_relations_effect = yes
                    }

                    if = {
                        limit = {
                            NOT = { exists = scope:awow_mother.dynasty}
                            exists = scope:awow_other_mother.dynasty
                        }
                        set_house = scope:awow_other_mother.house
                    }
                }
            }

            create_character_memory = {
                type = child_born
                participants = {
                    child = scope:childsr
                }
            }
            # debug_log = "AWOW SS-PREG: another child borned"
        }

        if = {
            limit = { has_character_flag = awow_pregnancy_will_be_triplets_flag }
            create_character = {
                age = 0
                random_traits = no
                culture = root.culture
                faith = root.faith
                dynasty = inherit
                location = root.location
                mother = root
                gender_female_chance = var:awow_baby_female_chance
                save_scope_as = childssr
                after_creation = {
                    if = {
                        limit = { scope:awow_mother = { is_ruler = yes } }
                        scope:awow_mother = { add_courtier = scope:childssr }
                    }
                    else_if = {
                        limit = {
                            exists = scope:awow_other_mother
                            scope:awow_other_mother = { is_ruler = yes }
                        }
                        scope:awow_other_mother = { add_courtier = scope:childssr }
                    }
                    else = {
                        scope:awow_mother = { add_courtier = scope:childssr }
                    }

                    scope:childssr = {
                        awow_assign_scripted_relations_effect = yes
                    }

                    if = {
                        limit = {
                            NOT = { exists = scope:awow_mother.dynasty}
                            exists = scope:awow_other_mother.dynasty
                        }
                        set_house = scope:awow_other_mother.house
                    }
                }
            }

            create_character_memory = {
                type = child_born
                participants = {
                    child = scope:childssr
                }
            }
            # debug_log = "AWOW SS-PREG: many childs born"
        }

        remove_trait = pregnant
        remove_variable = awow_pregnancy_active
        remove_character_flag = awow_pregnancy_will_be_twins_flag
        remove_character_flag = awow_pregnancy_will_be_triplets_flag
        carn_end_custom_pregnancy_effect = yes

        # debug_log = "AWOW SS-PREG: removed traits and flags child borned"

        ######### REDO EVENT POPUPS #########
        if = {
            limit = { is_ai = no }
            scope:child = {
                trigger_event = {
                    id = awow_pregnancy.0002
                }
            }
        }

        if = {
            limit = {
                exists = scope:awow_other_mother
                scope:awow_other_mother = {
                    is_ai = no
                    is_alive = yes
                }
            }
            scope:awow_other_mother = {
                trigger_event = {
                    id = awow_pregnancy.0003
                }
            }
        }
        clear_saved_scope = awow_other_mother
        # debug_log = "AWOW SS-PREG: not ai popups sent"
    }
}

# 0002 - Visible birth announcement for a female player mother (carrying mother).
awow_pregnancy.0002 = {
    type = character_event
    hidden = no
    theme = pregnancy

    trigger = {
        is_alive = yes
        is_female = yes
    }

    title = awow_pregnancy_0002_t
    desc = awow_pregnancy_0002_d

    option = {
        name = awow_pregnancy_0002_ok
        trigger = { always = yes }
    }
}

# 0003 - Visible birth announcement for the non-carrying player mother (co-mother).

# 0010 - Preg Pulse scheduler
awow_pregnancy.0010 = {
    type = character_event
    hidden = yes

    immediate = {
        trigger_event = {
            id = awow_pregnancy.0020
            days = 369
        }
    }
}

# 0020 - Preg Pulse
awow_pregnancy.0020 = {
    type = character_event
    hidden = yes

    immediate = {
        every_living_character = {
            awow_pregnancy_pulse_effect = yes
        }
        trigger_event = {
            id = awow_pregnancy.0010
        }
    }
}

awow_pregnancy.0030 = {

    immediate = {
        awow_start_same_sex_pregnancy_effect = yes
    }
}

awow_pregnancy.1000 = {
    type = character_event
    hidden = yes

    trigger = {
        is_alive = yes
    }

    immediate = {
        awow_convert_vanilla_relations_to_scripted_effect = yes
    }
}
