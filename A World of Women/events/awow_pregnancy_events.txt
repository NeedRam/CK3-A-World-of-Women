namespace = awow_pregnancy

# 0001 - Scripted birth for a same-sex (female-female) pregnancy.
# Fired on the pregnant mother via Carnalitas's carn_on_custom_pregnancy_stage_change
# hook 280 days after the pregnancy started. Creates a new FEMALE child with both women
# as parents. No male is ever involved.
awow_pregnancy.0001 = {
    type = character_event
    hidden = yes

    trigger = {
        is_alive = yes
        has_trait = pregnant
        exists = var:awow_other_mother_id
    }

    immediate = {
        debug_log = "AWOW SS-PREG: BIRTH event fired"
        # Resolve the other mother (may have died during the pregnancy).
        character = var:awow_other_mother_id
        save_scope_as = awow_other_mother

        # Compute the female-percentage for the child from the active game rule.
        # If the other mother is gone, default to 100%% (this is a world of women).
        if = {
            limit = { exists = scope:awow_other_mother }
            awow_baby_female_chance_value = yes
        }
        else = {
            set_variable = { name = awow_baby_female_chance value = 100 }
        }

        debug_log = "AWOW SS-PREG: BIRTH other_mother resolved"
        debug_log_scopes = no

        # Birth: both women are parents. Female father is data-only and safe.
        if = {
            limit = { exists = scope:awow_other_mother }
            create_character = {
                age = 0
                random_traits = no
                culture = root.culture
                faith = root.faith
                dynasty = root.dynasty
                employer = root.employer
                mother = root
                father = scope:awow_other_mother
                gender_female_chance = var:awow_baby_female_chance
                save_scope_as = child
            }
            debug_log = "AWOW SS-PREG: BIRTH created child with both mothers"
            debug_log_scopes = no
        }
        else = {
            create_character = {
                age = 0
                random_traits = no
                culture = root.culture
                faith = root.faith
                dynasty = root.dynasty
                employer = root.employer
                mother = root
                gender_female_chance = var:awow_baby_female_chance
                save_scope_as = child
            }
            debug_log = "AWOW SS-PREG: BIRTH created child mother only (other dead)"
            debug_log_scopes = no
        }

        # Link the child to the other mother explicitly (idempotent if she exists).
        if = {
            limit = { exists = scope:awow_other_mother }
            scope:child = { set_father = scope:awow_other_mother }
        }

        # Birth memory so the event shows up in the character's timeline.
        create_character_memory = {
            type = child_born
            participants = {
                child = scope:child
            }
        }

        # Cleanup: remove the visible trait and Carnalitas's custom-pregnancy state.
        remove_trait = pregnant
        remove_variable = awow_other_mother_id
        carn_end_custom_pregnancy_effect = yes

        # Announce to the player only if the mother is the player's character
        # (rare: a female player). AI mothers get no popup.
        if = {
            limit = { is_ai = no }
            scope:child = {
                trigger_event = {
                    id = awow_pregnancy.0002
                }
            }
        }
    }
}

# 0002 - Visible birth announcement for a female player mother.
awow_pregnancy.0002 = {
    type = character_event
    hidden = no

    trigger = {
        is_alive = yes
        is_female = yes
    }

    title = awow_pregnancy_0002_t
    desc = awow_pregnancy_0002_d

    option = {
        name = awow_pregnancy_0002_ok
        trigger = { always = yes }
    }
}

# 0003 - Monthly scheduler pulse (self-rescheduling). Iterates all living AI women and
# runs the same-sex conception roll. Started once at game start; reschedules every 30 days.
awow_pregnancy.0003 = {
    type = character_event
    hidden = yes

    immediate = {
        debug_log = "AWOW SS-PREG: scheduler pulse 0003 fired"
        awow_monthly_pregnancy_tick_effect = yes
    }
}
