namespace = awow

# Startup setup event - runs immediately, sets up variables and schedules first cull
awow.1000 = {
	type = character_event
	hidden = yes
	
	immediate = {
		# Set initial cull frequency to 6 months
		set_global_variable = {
			name = awow_culling_frequency
			value = 6
		}
		
		# Initialize months since cull counter
		set_global_variable = {
			name = awow_months_since_cull
			value = 0
		}
		
		# Initialize cull count so the tooltip always has a value to read
		set_global_variable = {
			name = awow_cull_count
			value = 0
		}
		
		# Schedule first cull in 30 days
		trigger_event = {
			id = awow.1001
			days = 30
		}
	}
}

# Initial cull - The Vanishing (with narrative popup)
# Fires 30 days after game start
awow.1001 = {
	type = character_event

	title = awow.1001.title
	desc = awow.1001.desc
	
	theme = death
	
	left_portrait = root
	
	immediate = {
		# Initialize cull count
		set_global_variable = {
			name = awow_cull_count
			value = 0
		}
		
		# Perform the culling
		awow_perform_culling = yes
		
		# Mark that initial cull is done
		set_global_variable = {
			name = awow_initial_cull_done
			value = yes
		}
		
		# Schedule the recurring monthly pulse (only once, from the first player to reach here)
		if = {
			limit = {
				NOT = { has_global_variable = awow_pulse_scheduled }
			}
			set_global_variable = {
				name = awow_pulse_scheduled
				value = yes
			}
			trigger_event = {
				id = awow.1002
				days = 30
			}
		}
	}
	
	option = {
		name = awow.1001.a
	}
}

# Recurring monthly culling pulse
# Runs every ~30 days, checks if a cull is due, and sends a bottom-right alert instead of a popup.
awow.1002 = {
	type = character_event
	hidden = yes
	
	immediate = {
		# Advance the cycle counter
		change_global_variable = {
			name = awow_months_since_cull
			add = 1
		}
		
		# Check if it's time to cull
		if = {
			limit = {
				global_var:awow_months_since_cull >= global_var:awow_culling_frequency
			}
			# Perform the world-wide cull exactly once
			awow_perform_culling = yes
			
			# Show a single bottom-right alert to each human player
			every_player = {
				send_interface_message = {
					type = awow_culling_message
				}
			}
		}
		
		# Schedule the next pulse in ~30 days
		trigger_event = {
			id = awow.1002
			days = 30
		}
	}
	
	option = {
		name = awow.1002.a
	}
}

# Birth gender assignment event
# Fires on_pregnancy, checks game rules to potentially set the unborn child's gender.
awow.0020 = {
	type = character_event
	hidden = yes
	
	immediate = {
		if = {
			limit = { is_ai = no }
			# Player birth - check player rule
			if = {
				limit = { has_game_rule = awow_birth_gender_player_67 }
				random = { chance = 67 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_player_75 }
				random = { chance = 75 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_player_80 }
				random = { chance = 80 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_player_90 }
				random = { chance = 90 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_player_95 }
				random = { chance = 95 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_player_99 }
				random = { chance = 99 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_player_100 }
				set_pregnancy_gender = female
			}
		}
		else = {
			# Non-player birth - check world rule
			if = {
				limit = { has_game_rule = awow_birth_gender_67 }
				random = { chance = 67 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_75 }
				random = { chance = 75 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_80 }
				random = { chance = 80 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_90 }
				random = { chance = 90 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_95 }
				random = { chance = 95 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_99 }
				random = { chance = 99 set_pregnancy_gender = female }
			}
			else_if = {
				limit = { has_game_rule = awow_birth_gender_100 }
				set_pregnancy_gender = female
			}
		}
	}
}
