namespace = awow

# Startup setup event - runs immediately, sets up variables and schedules first cull
event = {
	id = awow.1000
	hide_window = yes
	
	immediate = {
		# Set initial cull frequency to 6 months
		set_global_variable = {
			name = awow_culling_frequency
			value = 6
		}
		
		# Initialize months since cull counter
		set_global_variable = {
			name = awow_months_since_cull
			value = 0
		}
		
		# Schedule first cull in 30 days
		trigger_event = {
			id = awow.1001
			days = 30
		}
	}
}

# Initial cull - The Vanishing (with narrative popup)
# Fires 30 days after game start
event = {
	id = awow.1001
	
	title = awow.1001.title
	desc = awow.1001.desc
	
	theme = great_pox
	
	left_portrait = root
	
	immediate = {
		# Initialize cull count
		set_global_variable = {
			name = awow_cull_count
			value = 0
		}
		
		# Perform the culling
		awow_perform_culling = yes
		
		# Mark that initial cull is done
		set_global_variable = {
			name = awow_initial_cull_done
			value = yes
		}
	}
	
	option = {
		name = awow.1001.a
		custom_tooltip = awow_cull_count_tooltip
	}
}

# Birth gender reassignment event
# Fires on_childbirth, checks game rules to potentially change child's gender
event = {
	id = awow.0020
	hide_window = yes
	
	immediate = {
		# Check if the child's mother is the player
		if = {
			limit = {
				mother = { is_player = yes }
			}
			# Player birth - check player rule
			if = {
				limit = {
					game_rule = awow_birth_gender_player_67
				}
				if = {
					limit = { random_chance < 0.67 }
					set_character_gender = female
				}
			}
			else_if = {
				limit = {
					game_rule = awow_birth_gender_player_75
				}
				if = {
					limit = { random_chance < 0.75 }
					set_character_gender = female
				}
			}
