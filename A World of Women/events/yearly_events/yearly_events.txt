namespace = yearly

#Conclusion
yearly.0029 = {
    type = character_event
    title = yearly.0029.t
    desc = {
        desc = yearly.0029.conclusion_intro
        #What score did we have?
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:result = flag:bad_result
                }
                desc = {
                    random_valid = {
                        desc = yearly.0029.refurbishment_bad.1
                        desc = yearly.0029.refurbishment_bad.2
                        desc = yearly.0029.refurbishment_bad.3
                    }
                    random_valid = {
                        desc = yearly.0029.edibles_bad.1
                        desc = yearly.0029.edibles_bad.2
                        desc = yearly.0029.edibles_bad.3
                    }
                }
            }
            triggered_desc = {
                trigger = {
                    scope:result = flag:average_result
                }
                desc = {
                    random_valid = {
                        desc = yearly.0029.refurbishment_average.1
                        desc = yearly.0029.refurbishment_average.2
                        desc = yearly.0029.refurbishment_average.3
                    }
                    random_valid = {
                        desc = yearly.0029.edibles_average.1
                        desc = yearly.0029.edibles_average.2
                        desc = yearly.0029.edibles_average.3
                    }
                }
            }
            triggered_desc = {
                trigger = {
                    scope:result = flag:good_result
                }
                desc = {
                    random_valid = {
                        desc = yearly.0029.refurbishment_good.1
                        desc = yearly.0029.refurbishment_good.2
                        desc = yearly.0029.refurbishment_good.3
                    }
                    random_valid = {
                        desc = yearly.0029.edibles_good.1
                        desc = yearly.0029.edibles_good.2
                        desc = yearly.0029.edibles_good.3
                    }
                }
            }
        }
        #Had bard?
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:result = flag:bad_result
                    scope:entertainment = flag:bard
                }
                desc = yearly.0029.entertainment_bard_bad
            }
            triggered_desc = {
                trigger = {
                    scope:result = flag:average_result
                    scope:entertainment = flag:bard
                }
                desc = yearly.0029.entertainment_bard_average
            }
            triggered_desc = {
                trigger = {
                    scope:result = flag:good_result
                    scope:entertainment = flag:bard
                }
                desc = yearly.0029.entertainment_bard_good
            }
        }
        #Had poet?
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:result = flag:bad_result
                    scope:entertainment = flag:poet
                }
                desc = yearly.0029.entertainment_poet_bad
            }
            triggered_desc = {
                trigger = {
                    scope:result = flag:average_result
                    scope:entertainment = flag:poet
                }
                desc = yearly.0029.entertainment_poet_average
            }
            triggered_desc = {
                trigger = {
                    scope:result = flag:good_result
                    scope:entertainment = flag:poet
                }
                desc = yearly.0029.entertainment_poet_good
            }
        }
        #Had jester?
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:result = flag:bad_result
                    scope:entertainment = flag:jester
                }
                desc = yearly.0029.entertainment_jester_bad
            }
            triggered_desc = {
                trigger = {
                    scope:result = flag:average_result
                    scope:entertainment = flag:jester
                }
                desc = yearly.0029.entertainment_jester_average
            }
            triggered_desc = {
                trigger = {
                    scope:result = flag:good_result
                    scope:entertainment = flag:jester
                }
                desc = yearly.0029.entertainment_jester_good
            }
        }
        desc = yearly.0029.conclusion_outro
    }
    theme = stewardship
    override_background = tavern
    left_portrait = {
        character = root
        animation = drink
    }
    right_portrait = {
        trigger = {
            NAND = {
                scope:entertainment = flag:jester
                scope:result = flag:bad_result
            }
        }
        character = scope:entertainer
        #Bard
        ##Bad
        triggered_animation = {
            trigger = {
                scope:entertainment = flag:bard
                scope:result = flag:bad_result
            }
            animation = crying
        }
        ##Average
        triggered_animation = {
            trigger = {
                scope:entertainment = flag:bard
                scope:result = flag:average_result
            }
            animation = lute_idle
        }
        ##Good
        triggered_animation = {
            trigger = {
                scope:entertainment = flag:bard
                scope:result = flag:good_result
            }
            animation = lute_active
        }
        #Poet
        ##Bad
        triggered_animation = {
            trigger = {
                scope:entertainment = flag:poet
                scope:result = flag:bad_result
            }
            animation = stress
        }
        ##Average
        triggered_animation = {
            trigger = {
                scope:entertainment = flag:poet
                scope:result = flag:average_result
            }
            animation = debating
        }
        ##Good
        triggered_animation = {
            trigger = {
                scope:entertainment = flag:poet
                scope:result = flag:good_result
            }
            animation = storyteller
        }
        #Jester
        ##Bad
        ### The animation isn't missing, the jester just doesn't turn up in the narrative

        ##Average
        triggered_animation = {
            trigger = {
                scope:entertainment = flag:jester
                scope:result = flag:average_result
            }
            animation = eyeroll
        }
        ##Good
        triggered_animation = {
            trigger = {
                scope:entertainment = flag:jester
                scope:result = flag:good_result
            }
            animation = dancing
        }
    }

    trigger = {
        is_alive = yes
        is_available_adult = yes
        scope:county.holder = root
        NOT = {	has_trait = blind }
    }
    on_trigger_fail = { remove_variable = inn_goodness_score }

    immediate = {
        if = {
            limit = {
                var:inn_goodness_score <= 4
            }
            save_scope_value_as = {
                name = result
                value = flag:bad_result
            }
        }
        else_if = {
            limit = {
                var:inn_goodness_score >= 8
            }
            save_scope_value_as = {
                name = result
                value = flag:good_result
            }
        }
        else = {
            save_scope_value_as = {
                name = result
                value = flag:average_result
            }
        }

        #Create a bard
        if = {
            limit = {
                scope:entertainment = flag:bard
            }
            create_character = {
                location = root.location
                template = bard_template
                gender_female_chance = 100
                culture = scope:county.culture
                faith = scope:county.faith
                save_scope_as = entertainer
            }
        }
        #Create a poet
        else_if = {
            limit = {
                scope:entertainment = flag:poet
            }
            create_character = {
                location = root.location
                template = ep3_poet_template
                culture = scope:county.culture
                faith = scope:county.faith
                save_scope_as = entertainer
            }
        }
        #Create a jester
        else = {
            create_character = {
                location = root.location
                template = jester_template
                culture = scope:county.culture
                faith = scope:county.faith
                save_scope_as = entertainer
            }
        }
        remove_variable = inn_goodness_score
    }

    #Ending
    option = {
        name = yearly.0029.a
        if = {
            limit = {
                scope:result = flag:bad_result
            }
            capital_county.title_province = {
                add_province_modifier = {
                    modifier = innkeeper_bad_result
                    years = 40
                }
            }
            scope:entertainer = {
                if = {
                    limit = {
                        NOT = {
                            scope:entertainment = flag:jester
                        }
                    }
                    add_stress = 100
                }
            }
            add_stewardship_lifestyle_xp = minor_lifestyle_xp
        }
        else_if = {
            limit = {
                scope:result = flag:average_result
            }
            capital_county.title_province = {
                add_province_modifier = {
                    modifier = innkeeper_average_result
                    years = 40
                }
            }
            add_courtier = scope:entertainer
            add_stewardship_lifestyle_xp = major_lifestyle_xp
        }
        else = {
            limit = {
                scope:result = flag:good_result
            }
            capital_county.title_province = {
                add_province_modifier = {
                    modifier = innkeeper_good_result
                    years = 40
                }
            }
            add_courtier = scope:entertainer
            add_stewardship_lifestyle_perk_points = 1
        }
        stress_impact = {
            greedy = miniscule_stress_impact_gain
            gregarious = miniscule_stress_loss
        }
        ai_chance = {
            base = 1
            ai_value_modifier = {
                ai_sociability = 1
                ai_greed = -1
            }
        }
    }
    after = {
        scope:entertainer ?= {
            if = {
                limit = {
                    root = { is_ai = yes }
                    NOT = { is_courtier_of = root }
                }
                silent_disappearance_effect = yes
            }
        }
    }
}
