	category = interaction_category_uncategorized
	icon = debug_chars
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	auto_accept = yes

	is_shown = {
		debug_only = yes
		scope:actor = scope:recipient
	}

	on_auto_accept = {
        scope:actor = { save_scope_as = ufg_native_actor }
        set_variable = { name = ufg_acceptance_ok value = 1 }
        ufg_native_run_history_test_effect = yes
    }
}
