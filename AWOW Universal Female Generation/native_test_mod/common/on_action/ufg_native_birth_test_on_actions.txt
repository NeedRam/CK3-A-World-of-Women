# "Forced male" describes the child's selected gender, not the pregnant parent.
on_birth_child = {
	on_actions = {
		ufg_native_score_forced_male_child_birth_test
	}
}

ufg_native_score_forced_male_child_birth_test = {
	trigger = {
		scope:mother ?= {
			has_character_flag = ufg_native_forced_male_child_birth_test_mother
		}
	}

	effect = {
		if = {
			limit = { is_female = no }
			debug_log = "UFG_NATIVE_TEST_PASS: Forced-male child remained male at birth."
		}
		else = {
			debug_log = "UFG_NATIVE_TEST_FAIL: Forced-male child was changed to female at birth."
		}
		scope:mother = {
			remove_character_flag = ufg_native_forced_male_child_birth_test_mother
		}
	}
}
