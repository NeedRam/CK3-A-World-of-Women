303011_effect = {
	if = {
		limit = {
			exists = character:303011
			exists = character:303009
		}
		character:303011 = { save_scope_as = awow_child}
		character:303009 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
