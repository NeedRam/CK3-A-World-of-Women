144640_effect = {
	if = {
		limit = {
			exists = character:144640
			exists = character:28021
		}
		character:144640 = { save_scope_as = awow_child}
		character:28021 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
