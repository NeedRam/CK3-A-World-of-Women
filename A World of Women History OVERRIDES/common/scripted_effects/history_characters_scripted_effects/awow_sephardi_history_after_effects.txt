41709_effect = {
	if = {
		limit = {
			exists = character:41709
			exists = character:41713
		}
		character:41709 = { save_scope_as = awow_child}
		character:41713 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
