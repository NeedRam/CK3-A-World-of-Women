303302_effect = {
	if = {
		limit = {
			exists = character:303302
			exists = character:303301
		}
		character:303302 = { save_scope_as = awow_child}
		character:303301 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
