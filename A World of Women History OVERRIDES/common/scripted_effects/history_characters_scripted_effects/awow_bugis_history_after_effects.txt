bugis_cina_02_effect = {
	if = {
		limit = {
			exists = character:bugis_cina_02
			exists = character:bugis_cina_01
		}
		character:bugis_cina_02 = { save_scope_as = awow_child}
		character:bugis_cina_01 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
