ryukyuan_amami_2_effect = {
	if = {
		limit = {
			exists = character:ryukyuan_amami_2
			exists = character:ryukyuan_amami_1
		}
		character:ryukyuan_amami_2 = { save_scope_as = awow_child}
		character:ryukyuan_amami_1 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
