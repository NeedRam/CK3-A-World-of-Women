assign_history_filipino_other_mothers = {
	bisayan_zamboanga_11_fic_effect = yes
	bisayan_butuan_11_effect = yes
	bisayan_butuan_12_effect = yes
	bisayan_butuan_13_fic_effect = yes
	bisayan_panay_11_effect = yes
}

bisayan_zamboanga_11_fic_effect = {
	if = {
		limit = {
			exists = character:bisayan_zamboanga_11_fic
			exists = character:bisayan_zamboanga_10
		}
		character:bisayan_zamboanga_11_fic = { save_scope_as = awow_child}
		character:bisayan_zamboanga_10 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
