cham_bhrgu_01_effect = {
	if = {
		limit = {
			exists = character:cham_bhrgu_01
			exists = character:cham_bhrgu_00
		}
		character:cham_bhrgu_01 = { save_scope_as = awow_child}
		character:cham_bhrgu_00 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
