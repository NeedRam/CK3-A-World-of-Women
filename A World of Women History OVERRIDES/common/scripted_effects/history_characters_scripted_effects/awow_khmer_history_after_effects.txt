	if = {
		limit = {
			exists = character:khmer_kaundinya_2
			exists = character:khmer_kaundinya_1
		}
		character:khmer_kaundinya_2 = { save_scope_as = awow_child}
		character:khmer_kaundinya_1 = { save_scope_as = awow_other_mother}
		awow_assign_historic_father_as_other_mother = yes
		clear_saved_scope = awow_child
		clear_saved_scope = awow_other_mother
	}
}
